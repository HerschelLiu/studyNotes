  <div
    class="top-right-btn"
    :style="style">
    <el-row>

<script setup lang="ts">
  import cache from '@/plugins/cache'
  import type { TableShowColumns } from '@/types/api/common'

  const props = defineProps({
    /* 是否显示检索条件 */
    showSearch: {
      type: Boolean,
      default: true
    },
    /* 显隐列信息（数组格式、对象格式） */

      type: [Array, Object],

    /* 是否显示检索图标 */
    search: {

      default: true
    },

    showColumnsType: {
      type: String,
      default: "checkbox"
    },
    /* 右外边距 */
    gutter: {
      type: Number,
      default: 10
    },
    /* 列显隐状态记忆的 localStorage key（传入则启用记忆，不传则不记忆） */
    storageKey: {
      type: String,
      default: ""
    }
  })

  const emits = defineEmits(['update:showSearch', 'queryTable'])

  // 显隐数据
  const value = ref<number[]>([])
  // 弹出层标题
  const title = ref("显示/隐藏")
  // 是否显示弹出层
  const open = ref(false)

  const style = computed(() => {
    const ret: Record<string, string> = {}
    if (props.gutter) {
      ret.marginRight = `${props.gutter / 2}px`
    }
    return ret
  })

  // 是否全选/半选 状态
  const isChecked = computed({
    get: () => Array.isArray(props.columns) ? props.columns.every((col: TableShowColumns) => col.visible) : Object.values(props.columns).every((col) => (col as TableShowColumns).visible),
    set: () => {}
  })
  const isIndeterminate = computed(() => Array.isArray(props.columns) ? props.columns.some((col: TableShowColumns) => col.visible) && !isChecked.value : Object.values(props.columns).some((col) => (col as TableShowColumns).visible) && !isChecked.value)
  const transferData = computed(() => Array.isArray(props.columns) ? props.columns.map((item: any, index: number) => ({ key: index, label: item.label })) : Object.keys(props.columns).map((key, index) => ({ key: index, label: (props.columns as any)[key].label })))

  // 搜索
  const { proxy } = getCurrentInstance()!
  function toggleSearch(): void {
    let el: HTMLElement | null = proxy!.$el
    let formEl: HTMLElement | null = null
    while ((el = el!.parentElement) && el !== document.body) {
      if ((formEl = el.querySelector('.el-form'))) break
    }
    if (!formEl) return emits('update:showSearch', !props.showSearch)
    // 优先动画整个 search-wrapper（el-card），避免表单先消失、卡片后消失的两段式抖动
    const targetEl = formEl.closest('.search-wrapper') as HTMLElement | null
    animateSearch(targetEl || formEl, props.showSearch)
  }
  function animateSearch(el: HTMLElement, isHide: boolean): void {
    const DURATION = 260
    const TRANSITION = 'max-height 0.25s ease, opacity 0.2s ease'
    const clear = () => Object.assign(el.style, { transition: '', maxHeight: '', opacity: '', overflow: '' })
    Object.assign(el.style, { overflow: 'hidden', transition: '' })
    if (isHide) {
      Object.assign(el.style, { maxHeight: el.scrollHeight + 'px', opacity: '1', transition: TRANSITION })
      requestAnimationFrame(() => Object.assign(el.style, { maxHeight: '0', opacity: '0' }))
      setTimeout(() => { emits('update:showSearch', false); clear() }, DURATION)
    } else {
      emits('update:showSearch', true)
      nextTick(() => {
        Object.assign(el.style, { maxHeight: '0', opacity: '0' })
        requestAnimationFrame(() => requestAnimationFrame(() => {
          Object.assign(el.style, { transition: TRANSITION, maxHeight: el.scrollHeight + 'px', opacity: '1' })
        }))
        setTimeout(clear, DURATION)
      })
    }
  }
  // 刷新
  function refresh(): void {
    emits("queryTable")
  }

  // 右侧列表元素变化
  function dataChange(data: any[], _direction?: string, _movedKeys?: any[]): void {
    if (Array.isArray(props.columns)) {
      for (let item in props.columns) {
        const key = props.columns[item].key
        props.columns[item].visible = !data.includes(parseInt(key))
      }
    } else {
      Object.keys(props.columns).forEach((key, index) => {
        props.columns[key].visible = !data.includes(index)
      })
    }
    saveStorage()
  }

  // 打开显隐列dialog
  function showColumn(): void {
    open.value = true
  }

  // 如果传入了 storageKey，从 localStorage 恢复列显隐状态
  if (props.storageKey) {
    try {
      const saved = cache.local.getJSON(props.storageKey)
      if (saved && typeof saved === 'object') {
        if (Array.isArray(props.columns)) {
          props.columns.forEach((col: TableShowColumns, index: number) => {
            // 数组元素带 key 时按 key 恢复（CompTable 列路径），否则按数组下标
            const stateKey = (col as TableShowColumns & { key?: string }).key ?? index
            if (saved[stateKey] !== undefined) col.visible = saved[stateKey]
          })
        } else {
          Object.keys(props.columns).forEach(key => {
            if (saved[key] !== undefined) props.columns[key].visible = saved[key]
          })
        }
      }
    } catch (e) {}
  }
  if (props.showColumnsType == "transfer") {
    // transfer穿梭显隐列初始默认隐藏列
    if (Array.isArray(props.columns)) {
      for (let item in props.columns) {
        if (props.columns[item].visible === false) {
          value.value.push(parseInt(item))
        }
      }
    } else {
      Object.keys(props.columns).forEach((key, index) => {
        if (props.columns[key].visible === false) {
          value.value.push(index)
        }
      })
    }
  }

  // 单勾选
  function checkboxChange(event: any, key: string | number): void {
    if (Array.isArray(props.columns)) {
      // key 是 v-for 的索引，直接按索引定位
      const idx = Number(key)
      const col = props.columns[idx]
      if (col) {
        col.visible = event
      }
    } else {
      ;(props.columns as any)[key].visible = event
    }
    saveStorage()
  }

  // 切换全选/反选
  function toggleCheckAll(): void {
    const newValue = !isChecked.value
    if (Array.isArray(props.columns)) {
      props.columns.forEach((col: TableShowColumns) => (col.visible = newValue))
    } else {
      Object.values(props.columns).forEach((col) => ((col as TableShowColumns).visible = newValue))
    }
    saveStorage()
  }

  // 将当前列显隐状态持久化到 localStorage
  function saveStorage(): void {
    if (!props.storageKey) return
    try {
      let state: { [key: string]: boolean } = {}
      if (Array.isArray(props.columns)) {
        // 数组元素带 key 时按 key 存储（CompTable 列路径），否则按数组下标
        props.columns.forEach((col: TableShowColumns, index: number) => { state[(col as TableShowColumns & { key?: string }).key ?? index] = col.visible })
      } else {
        Object.keys(props.columns).forEach(key => { state[key] = props.columns[key].visible })
      }
      cache.local.setJSON(props.storageKey, state)
    } catch (e) {}
  }
</script>

```ts
import type { Component } from "vue";

/** 内置按钮语义，覆盖项目里最常见的工具栏操作 */
export type ToolbarActionKey =
    | "add"
    | "edit"
    | "del"
    | "batch"
    | "import"
    | "export"
    | "download"
    | string;

/** 单个工具栏按钮配置 */
export interface ToolbarButton {
    /** 动作标识，点击后随 action 事件回传（如 'add' / 'export'），也用于匹配内置默认样式 */
    key: ToolbarActionKey;
    /** 按钮文案；不传则取内置默认（如 add -> "新增"） */
    label?: string;
    /** el-button 的 type；不传则取内置默认 */
    type?: "primary" | "success" | "warning" | "danger" | "info" | "";
    /** 按钮图标名（Element Plus 图标组件名，如 'Plus'）或图标组件；不传则取内置默认 */
    icon?: string | Component;
    /** 是否 plain 风格，默认 true（对齐项目现状） */
    plain?: boolean;
    /** 按钮尺寸，默认 'default' */
    size?: "large" | "default" | "small";
    /** 是否禁用 */
    disabled?: boolean;
    /** 是否隐藏该按钮（用于按权限/状态动态控制） */
    hidden?: boolean;
    /** 权限标识，透传给 v-hasPermi 指令（旧范式常用），为空则不加限制 */
    perm?: string;
}

export interface TableToolbarProps {
    /** 按钮配置数组，按顺序渲染 */
    buttons?: ToolbarButton[];
    /** 是否显示右侧 right-toolbar（搜索开关/刷新/显隐列），默认 true */
    showRightToolbar?: boolean;
    /** 搜索区显隐状态，配合 v-model:showSearch */
    showSearch?: boolean;
    /** 透传给 right-toolbar 的列显隐配置 */
    columns?: any[] | Record<string, any>;
    /** right-toolbar 的右外边距 gutter，默认 10 */
    gutter?: number;
    /** right-toolbar 列显隐状态本地存储 key */
    storageKey?: string;
    /** 是否显示 right-toolbar 内的显隐列/搜索开关图标（透传 search），默认 true */
    showColumnSetting?: boolean;
}

export interface TableToolbarEmits {
    /** 点击任一按钮，回传该按钮的 key（如 'add' / 'export'） */
    (e: "action", key: ToolbarActionKey, button: ToolbarButton): void;
    /** 搜索区显隐切换，配合 v-model:showSearch */
    (e: "update:showSearch", value: boolean): void;
    /** right-toolbar 刷新按钮点击，等价于原 @queryTable */
    (e: "queryTable"): void;
}

/** 内置按钮的默认样式映射（label / type / icon） */
export const DEFAULT_BUTTON_PRESET: Record<
    string,
    { label: string; type: ToolbarButton["type"]; icon: string }
> = {
    add: { label: "新增", type: "primary", icon: "Plus" },
    edit: { label: "修改", type: "success", icon: "Edit" },
    del: { label: "删除", type: "danger", icon: "Delete" },
    batch: { label: "批量删除", type: "danger", icon: "Delete" },
    import: { label: "导入", type: "info", icon: "Upload" },
    export: { label: "导出", type: "success", icon: "Upload" },
    download: { label: "下载模板", type: "warning", icon: "Tickets" },
};

<script lang="ts" setup generic="T extends Record<string, any>, R extends Record<string, any>">
  import type { DefineCheckAllEmits } from '@/hooks/useCheckAll'
  import type { DefineListEmits } from '@/hooks/useList'
  import type { CompTableProps, CompTableSlots, TableColumn } from './types'

  import { useCheckAll } from '@/hooks/useCheckAll'
  import { useListAdaptor } from '@/hooks/useList'
  import { useTableHeight } from '@/hooks/useTable'
  import { useTempRefs } from '@/hooks/useTempRefs'
  import { compTableColumnsKey } from './CompTableColumn'
  import { createDynamicColumns } from './renderColumns'
  import { useColumnVisibility } from './useColumnVisibility'

  import TableToolbar from '@/components/TableToolbar/index.vue'

  defineOptions({ name: 'CompTable' })

  const props = withDefaults(defineProps<CompTableProps<T, R>>(), {
    selected: () => [],
    keys: '',
    storageKey: '',
    tableKey: 'default',
    showTableToolBar: true,
    showSearch: true,
    selectable: true,
    selectionWidth: 55,
    minWidth: 100,
    showSummary: false,
    summaryMethod: undefined,
    showRightToolbar: true,
    heightMode: 'fixed',
    index: false,
    defaultExpandAll: false,
    treeProps: () => ({ hasChildren: 'hasChildren', children: 'children', checkStrictly: false }),
    showToolBarSearch: true,
    required: false,
    bottomOffset: 20 + 20 + 32,
  })

  interface Emits
    extends Omit<DefineListEmits<T, R>, 'update:list'>,
      DefineCheckAllEmits<R> {
    /** 刷新按钮点击 */
    queryTable: []
    /** 搜索区显隐切换 */
    'update:showSearch': [val: boolean]
    /** List 模式回传整个 List；数组模式回传 items 数组 */
    'update:list': [value: List<T, R> | R[]]
  }
  const emit = defineEmits<Emits>()
  defineSlots<CompTableSlots<R>>()

  /** showSearch 双向绑定：TableToolbar 切换 → 透传给父组件 */
  const innerShowSearch = computed({
    get: () => props.showSearch,
    set: (val: boolean) => emit('update:showSearch', val),
  })

  // ---------- 表格高度 ----------
  const { ELTable } = useTempRefs('ELTable')
  const { tableHeight, refresh: refreshTableHeight } = useTableHeight(
    ELTable,
    props.bottomOffset
  )

  /**
   * 高度解析：
   * - 显式 height 优先，直接使用（不参与视口计算）
   * - fixed（默认）：height = 视口剩余高度
   * - max：max-height = maxHeight ?? 视口剩余高度
   * - auto：不限制高度
   */
  const resolvedHeight = computed<ID | undefined>(() => {
    if (props.height != null) return props.height
    return props.heightMode === 'fixed' ? tableHeight.value : undefined
  })
  const resolvedMaxHeight = computed<ID | undefined>(() => {
    if (props.height != null || props.heightMode !== 'max') return undefined
    return props.maxHeight ?? tableHeight.value
  })

  /** 搜索区显隐 / 高度策略变化后表格尺寸变化，需在新一帧布局完成后重新计算高度 */
  let heightRaf = 0
  const scheduleHeightRefresh = () => {
    cancelAnimationFrame(heightRaf)
    heightRaf = requestAnimationFrame(() => refreshTableHeight())
  }
  watch([() => props.showSearch, () => props.heightMode], scheduleHeightRefresh)
  onScopeDispose(() => cancelAnimationFrame(heightRaf))

  /** el-table row-key：优先 rowKey prop，回退 keys */
  const resolvedRowKey = computed(() => props.rowKey ?? (props.keys || undefined))

  // ---------- 列表数据 ----------
  /** List 模式下的空行数组（引用稳定，避免每次求值新建数组参与深度监听） */
  const EMPTY_ROWS: R[] = []

  /** 数组模式经 useListAdaptor 适配为 List；List 模式直接透传（两种模式都原地操作，不回写数组） */
  const adaptorList = useListAdaptor<R>(() => (Array.isArray(props.list) ? props.list : EMPTY_ROWS))

  const list = computed<List<T, R>>({
    get: () =>
      Array.isArray(props.list) ? ((adaptorList as unknown) as List<T, R>) : props.list,
    set: val => emit('update:list', val),
  })

  /** 跨页选择 v-model:selected + 全选 + 单行勾选 */
  const { isCheckAll, handleChecked } = useCheckAll(props.keys, list, props, { emit })

  const warnedTableKeys = new Set<string>()
  watchEffect(() => {
    if (!import.meta.env.DEV || !columnsMap || columnsMap.has(props.tableKey) || warnedTableKeys.has(props.tableKey)) return
    warnedTableKeys.add(props.tableKey)
    console.warn(`[CompTable] table-key "${props.tableKey}" 未在 useCompTable 中注册，已注册：${[...columnsMap.keys()].join(', ') || '(无)'}`)

  const { columnVisibility, dataColumns, indexColumnConfig, expandColumnConfig } = useColumnVisibility<R>({

    dataColumns,
    getTableProps: () => props,

import type { ComputedRef, VNode } from 'vue'
import type { DefaultRow } from 'element-plus/es/components/table/src/table/defaults'
import type { CellScope, CompTableProps, HeaderScope, TableColumn, TableColumnFormat } from './types'
import type { ColumnNodeSlots } from './CompTableColumn'
import type { FormatRaw } from '@/hooks/useFormatValue'

import { Fragment } from 'vue'
import { ElTableColumn } from 'element-plus'

import { useFormatValue } from '@/hooks/useFormatValue'
import { useRulesRequired } from '@/hooks/useRules'
import { RequiredHeader } from '@/hooks/useTable'

import CompPrice from '@/components/Price/index.vue'
import { resolveColPath } from './columnPath'
import { CompTableColumn } from './CompTableColumn'

/** 列渲染需要的表格级默认值（全局对齐 / 最小宽度 / 必填规则） */
type ColumnDefaults<R extends DefaultRow> = Pick<CompTableProps<any, R>, 'minWidth' | 'required' | 'align' | 'headerAlign'>

/** 按列格式化配置分派，保留 useFormatValue 各重载的参数约束 */
const formatCellValue = (raw: FormatRaw, format?: TableColumnFormat) => {
  if (!format) return useFormatValue(raw)
  if (format.type === 'enum') return useFormatValue(raw, 'enum', format.formatStr)
  if (format.type === 'dict') return useFormatValue(raw, 'dict', format.formatStr)
  return useFormatValue(raw, format.type, format.formatStr)
}

interface CreateDynamicColumnsOptions<R extends DefaultRow> {
  /** 已过滤的渲染列（分组 + 叶子） */
  dataColumns: ComputedRef<TableColumn<R>[]>
  /** 表格级 props（列未配置时取这里的默认值） */
  getTableProps: () => ColumnDefaults<R>
}

/**
 * 稳定列 key 生成器（按列身份 = 层级 + 路径）：
 * - 列显隐只增删自身，其余列的 key 不变，避免 el-table-column 整体重挂载
 *   （重挂载会重新注册列、丢失列配置并触发多次 layout）
 * - 身份重复（未按约定传 key）时追加序号，保证同一级 vnode key 唯一
 */
const createColumnKey = () => {
  const occurrences = new Map<string, number>()
  return (identity: string) => {
    const times = occurrences.get(identity) ?? 0
    occurrences.set(identity, times + 1)
    return times ? `${identity}#${times}` : identity
  }
}

/** 展平默认插槽里的 Fragment（在 Vue 3 中，Fragment（片段） 是一个内置的虚拟包装概念，它允许组件模板拥有多个根节点，而无需像 Vue 2 那样必须包裹在一个单一的 <div> 里） / 数组，拿到 <CompTableColumn> 标记的 VNode */
const flattenVNodes = (children: unknown): VNode[] => {
  if (Array.isArray(children)) return children.flatMap(item => flattenVNodes(item))
  if (children && typeof children === 'object') {
    const vnode = children as VNode
    if (vnode.type === Fragment) return flattenVNodes(vnode.children)
    if ('type' in vnode) return [vnode]
  }
  return []
}

/**
 * 创建动态数据列函数式组件：
 * - 支持 children 递归渲染多级表头
 * - 单元格优先级：<CompTableColumn>#default > price > format
 * - 表头优先级：<CompTableColumn>#header > renderHeader=true > label
 *
 * 必须在 setup 同步执行期间调用（内部使用 useSlots）
 */
export const createDynamicColumns = <R extends Record<string, any>>({ dataColumns, getTableProps }: CreateDynamicColumnsOptions<R>) => {
  const slots = useSlots()
  const isRequiredByRules = useRulesRequired(() => getTableProps().required)

  /** <CompTableColumn> 没有提供任何插槽时的告警去重 */
  const warnedEmptyColumnNodes = new Set<string>()
  const warnEmptyColumnNode = (prop: string) => {
    if (!import.meta.env.DEV || warnedEmptyColumnNodes.has(prop)) return
    warnedEmptyColumnNodes.add(prop)
    console.warn(`[CompTable] 列 "${prop}" 的 <CompTableColumn> 没有 #header/#default 插槽，已忽略。`)
  }

  /** 收集默认插槽里所有 <CompTableColumn prop="..."> 及其 #header/#default 插槽 */
  const collectColumnNodes = (): Map<string, ColumnNodeSlots<R>> => {
    const map = new Map<string, ColumnNodeSlots<R>>()
    flattenVNodes(slots.default?.()).forEach(vnode => {
      if (vnode.type !== CompTableColumn) return
      const prop = vnode.props?.prop
      if (prop == null) return

      const node: ColumnNodeSlots<R> = {}
      const raw = vnode.children
      if (raw && typeof raw === 'object' && !Array.isArray(raw)) {
        const rawSlots = raw as Record<string, unknown>
        node.header = rawSlots.header as ColumnNodeSlots<R>['header']
        node.default = rawSlots.default as ColumnNodeSlots<R>['default']
      } else if (Array.isArray(raw)) {
        // 直接写在 <CompTableColumn> 里的内容视为单元格插槽
        node.default = () => raw
      }

      if (!node.header && !node.default) warnEmptyColumnNode(String(prop))
      map.set(String(prop), node)
    })
    return map
  }

  /** 叶子列单元格（优先级：<CompTableColumn>#default > price > format） */
  const renderLeafCell = (col: TableColumn<R>, scope: CellScope<R>, cellSlot?: ColumnNodeSlots<R>['default']) => {
    // el-table-column 渲染时会以 $index=-1 探测一次 default 插槽来收集子列，此时不渲染单元格
    if (scope.$index < 0) return null
    if (cellSlot) return cellSlot({ row: scope.row, column: scope.column, $index: scope.$index })
    if (col.price) {
      return h(CompPrice, {
        price: scope.row[col.prop as keyof R],
        ...(col.priceConfig ?? {})
      })
    }
    return formatCellValue(scope.row[col.prop as keyof R] as FormatRaw, col.format)
  }

  /** 表头：<CompTableColumn>#header 优先；其次 renderHeader=true 必填红星 */
  const renderColumnHeader = (col: TableColumn<R>, scope: HeaderScope<R>, headerNode?: ColumnNodeSlots<R>['header']) => {
    if (headerNode) return headerNode({ column: scope.column, $index: scope.$index })
    if (col.renderHeader === true) {
      return h(RequiredHeader, {
        label: col.label,
        required: isRequiredByRules(String(col.prop))
      })
    }
  }

  /** 递归渲染单列：有 children 渲染为分组表头，否则为叶子数据列 */
  const renderColumn = (
    col: TableColumn<R>,
    columnNodes: Map<string, ColumnNodeSlots<R>>,
    defaults: ColumnDefaults<R>,
    keyOf: (identity: string) => string,
    parentPath: string,
    depth: number
  ): VNode => {
    const path = resolveColPath(col, parentPath)
    const node = columnNodes.get(String(col.prop))
    const cellSlot = node?.default
    const headerNode = node?.header
    // 父列（有 children）也写了 #default 时降级为叶子列渲染插槽，children 被忽略
    const hasChildren = !!col.children?.length && !cellSlot

    const columnSlots: Record<string, (scope: any) => unknown> = {
      default: hasChildren
        ? () => col.children!.map(child => renderColumn(child, columnNodes, defaults, keyOf, path, depth + 1))
        : (scope: CellScope<R>) => renderLeafCell(col, scope, cellSlot)
    }
    if (headerNode || col.renderHeader === true) {
      columnSlots.header = (scope: HeaderScope<R>) => renderColumnHeader(col, scope, headerNode)
    }

    return h(
      ElTableColumn,
      {
        // 层级 + 路径：显隐只增删自身时其余列 key 不变；层级前缀避免根列与同名分组撞 key
        key: keyOf(`${depth}|${path}`),
        prop: hasChildren ? undefined : String(col.prop),
        label: col.label,
        width: col.width,
        minWidth: hasChildren ? col.minWidth : col.minWidth ?? defaults.minWidth,
        fixed: col.fixed,
        align: col.align ?? defaults.align,
        headerAlign: col.headerAlign ?? defaults.headerAlign ?? col.align ?? defaults.align,
        showOverflowTooltip: col.showOverflowTooltip,
        // 有自定义表头插槽时不能再传 renderHeader 函数，否则 Element Plus 会优先用函数
        renderHeader: !headerNode && typeof col.renderHeader === 'function' ? col.renderHeader : undefined
      },
      columnSlots
    )
  }

  /** 函数式组件：渲染期求值，自动追踪 dataColumns 显隐变化与 <CompTableColumn> 子节点变化 */
  return () => {
    const columnNodes = collectColumnNodes()
    const defaults = getTableProps()
    const keyOf = createColumnKey()
    return dataColumns.value.map(col => renderColumn(col, columnNodes, defaults, keyOf, '', 0))
  }
}

```ts
import type { TableColumn } from './types'

import { useLocalStorage } from '@vueuse/core'

import { resolveColPath } from './columnPath'

/** 全角空格用于显隐面板子列缩进（RightToolbar 不支持层级结构） */
const LEVEL_INDENT = '　'

/** 显隐面板条目 */
interface VisItem {
  key: string
  label: string
  visible: boolean
  level: number
  parentKey?: string
}

/**
 * 递归扁平化列结构为显隐列表：
 * - disabled 列跳过；分组子列全 disabled 时整组跳过
 * - 子列 label 加缩进前缀表达层级
 */
const flatten = <R>(cols: TableColumn<R>[], level = 0, parentKey?: string, parentPath = ''): VisItem[] =>
  cols.flatMap(col => {
    if (col.disabled) return []
    const key = resolveColPath(col, parentPath)
    const label = level ? `${LEVEL_INDENT.repeat(level)}└ ${col.label}` : col.label
    if (col.children?.length) {
      const children = flatten(col.children, level + 1, key, key)
      return children.length ? [{ key, label, visible: col.visible !== false, level, parentKey }, ...children] : []
    }
    return [{ key, label, visible: col.visible !== false, level, parentKey }]
  })

/** 查找首个匹配的叶子列 */
const findLeaf = <R>(cols: TableColumn<R>[], fn: (c: TableColumn<R>) => boolean): TableColumn<R> | undefined => {
  for (const col of cols) {
    if (col.children?.length) {
      const hit = findLeaf(col.children, fn)
      if (hit) return hit
    } else if (fn(col)) {
      return col
    }
  }
}

interface Options<R extends Record<string, any>> {
  /** 列配置（getter，保持对 useCompTable 注入列 / 动态列结构的响应） */
  getColumns: () => TableColumn<R>[]
  /** 显隐持久化 localStorage key，空串不持久化 */
  getStorageKey: () => string
}

/**
 * 多级表头列显隐（父子级联）。支持运行时列结构变化（如切换预算任务），
 * 结构变化时重建条目并按 key 保留各列已有显隐状态
 */
export const useColumnVisibility = <R extends Record<string, any>>({ getColumns, getStorageKey }: Options<R>) => {
  /**
   * 显隐持久化状态（useLocalStorage，JSON 序列化与旧格式兼容）：
   * - 按列路径 key 存储（key 以 '/' 开头，永不纯数字）
   * - 旧格式为纯数字下标 key，读取时按下标兜底，persist 时迁移为新格式
   * - writeDefaults: false：无已存数据时不写入默认值
   */
  const savedState = useLocalStorage<Record<string, boolean>>(computed(() => getStorageKey()), {}, { writeDefaults: false })

  /** 存储中是否读到旧下标格式（挂载时需要迁移为新的路径格式） */
  let appliedLegacy = false

  /** 把存储的显隐状态应用到条目列表 */
  const applySaved = (list: VisItem[]) => {
    if (!getStorageKey()) return
    const saved = savedState.value
    const keys = Object.keys(saved)
    if (!keys.length) return
    const legacy = keys.every(k => /^\d+$/.test(k))
    if (legacy) appliedLegacy = true
    list.forEach((item, idx) => {
      const value = legacy ? saved[idx] : saved[item.key]
      if (typeof value === 'boolean') item.visible = value
    })
  }

  /** 显隐条目列表（结构变化时原地 splice 重建，保留各列显隐状态） */
  const items = reactive(flatten(getColumns()))
  // 初始化即恢复显隐（异步列场景下此时为空转，列到达后由 rebuild 恢复）
  applySaved(items)

  /** key -> 自身可见状态（渲染过滤用） */
  const visMap = computed(() => new Map(items.map(i => [i.key, i.visible])))
  /** key -> 条目（级联时查父节点） */
  const itemMap = computed(() => new Map(items.map(i => [i.key, i])))
  /** 父 key -> 直接子条目列表（级联用） */
  const childrenMap = computed(() => {
    const map = new Map<string, VisItem[]>()
    items.forEach(i => {
      if (!i.parentKey) return
      const list = map.get(i.parentKey) ?? []
      list.push(i)
      map.set(i.parentKey, list)
    })
    return map
  })

  /** 某分组下全部后代 */
  const descendants = (key: string): VisItem[] => (childrenMap.value.get(key) ?? []).flatMap(kid => [kid, ...descendants(kid.key)])

  /** 某分组下全部叶子后代 */
  const leafDescendants = (key: string): VisItem[] => descendants(key).filter(d => !childrenMap.value.get(d.key)?.length)

  /** 持久化（空列不写，避免覆盖异步列到达前的已存状态） */
  const persist = () => {
    if (!getStorageKey() || !items.length) return
    const state: Record<string, boolean> = {}
    items.forEach(i => (state[i.key] = i.visible))
    savedState.value = state
  }

  // ---------- 程序化写入守卫 ----------
  // ready=false：挂载时 RightToolbar 从 localStorage 恢复显隐，尚未开始接收用户操作
  // syncing    ：正在写子孙 / 祖先 visible 或重建条目，watcher 据此跳过级联
  let ready = false
  let syncing = false

  /** 执行程序化写入；watcher 为 flush: 'sync'，写入瞬间即被 syncing 拦下 */
  const silentWrite = (write: () => void) => {
    syncing = true
    try {
      write()
    } finally {
      syncing = false
    }
  }

  /** 对用户切换的列执行父子级联 */
  const cascade = (changed: VisItem[]) => {
    silentWrite(() => {
      changed.forEach(item => {
        if (childrenMap.value.get(item.key)?.length) {
          // 分组：整组子孙跟随
          descendants(item.key).forEach(d => {
            d.visible = item.visible
          })
          return
        }
        // 叶子：自底向上，分组 visible = 子树是否存在可见叶子
        let pKey = item.parentKey
        while (pKey) {
          const parent = itemMap.value.get(pKey)
          if (parent) parent.visible = leafDescendants(parent.key).some(l => l.visible)
          pKey = parent?.parentKey
        }
      })
    })
    persist()
  }

  /**
   * 列结构变化时重建条目，按 key 保留显隐状态。
   * 用原地 splice 替换，保持 reactive 引用不变（RightToolbar 仍绑定同一数组）
   */
  const rebuild = () => {
    const prevVisible = new Map(items.map(i => [i.key, i.visible]))
    const next = flatten(getColumns())
    // 内存中无记录的新列（如异步列首次到达），从存储中按 key 恢复
    applySaved(next.filter(i => !prevVisible.has(i.key)))
    silentWrite(() => {
      items.splice(0, items.length, ...next.map(i => ({ ...i, visible: prevVisible.get(i.key) ?? i.visible })))
    })
    // 结构变化后回写存储，避免已删除列 / 旧下标格式残留
    persist()
  }
  // 深度监听列配置：结构相同则不触发，结构变化才重建（Vue 内部做深度比较）
  watch(getColumns, rebuild, { deep: true })

  /**
   * 挂载后：按叶子状态归并分组显隐（修复历史存储中"父隐藏/子显示"的不一致），
   * 之后才开始接收用户操作。仅在状态确实变化或需要迁移旧格式时回写存储，
   * 避免每次挂载都产生一次无意义的 localStorage 写入
   */
  onMounted(() =>
    nextTick(() => {
      let changed = false
      silentWrite(() => {
        items.forEach(i => {
          if (!childrenMap.value.get(i.key)?.length) return
          const visible = leafDescendants(i.key).some(l => l.visible)
          if (visible === i.visible) return
          i.visible = visible
          changed = true
        })
      })
      if (changed || appliedLegacy) persist()
      ready = true
    })
  )

  /**
   * 监听各列 visible 变化：仅处理用户主动切换（级联 / 恢复 / 重建均已同步置位 syncing）。
   * flush: 'sync' 让 silentWrite 能拦截本次写入引发的回调，无需跨 tick 复位标记
   */
  watch(
    () => items.map(i => [i.key, i.visible] as const),
    (_curr, prev) => {
      if (!ready || syncing) return
      const prevVisible = new Map(prev)
      const changed = items.filter(i => prevVisible.get(i.key) !== i.visible)
      if (changed.length) cascade(changed)
    },
    { flush: 'sync' }
  )

  // ---------- 渲染过滤 ----------
  /**
   * 递归过滤多级表头：
   * - 叶子生效可见 = 自身可见 && 所有祖先可见（父隐藏级联隐藏全部子列）
   * - disabled / prop='index' 叶子剔除（序号列单独静态渲染）
   * - 分组无生效子列时整组剔除
   */
  const filterTree = (cols: TableColumn<R>[], parent = '', parentVis = true): TableColumn<R>[] =>
    cols.flatMap(col => {
      if (col.disabled) return []
      const key = resolveColPath(col, parent)
      const own = visMap.value.get(key) ?? col.visible !== false
      const eff = parentVis && own
      if (col.children?.length) {
        const ch = filterTree(col.children, key, eff)
        return ch.length ? [{ ...col, children: ch }] : []
      }
      if (col.prop === 'index' || col.prop === 'expand') return []
      return eff ? [col] : []
    })

  const dataColumns = computed(() => filterTree(getColumns()))
  const indexColumnConfig = computed(() => findLeaf(getColumns(), c => c.prop === 'index'))
  const expandColumnConfig = computed(() => findLeaf(getColumns(), c => c.prop === 'expand'))

  return { columnVisibility: items, dataColumns, indexColumnConfig, expandColumnConfig }
}

export function useCompTable<R extends Record<string, any>>(
  columns: MaybeRefOrGetter<TableColumn<R>[]>,
  key = 'default',
) {
  const instance = getCurrentInstance()
  if (!instance) throw new Error('[CompTable] useCompTable() 必须在 setup 中调用')

>
> - useRulesRequired外层函数 接收`rules` ， 内层函数 接收`key` 。两者参数不同，所以不冲突、不报错。
> - `rules` 被内层函数通过 闭包 捕获，不需要第二次传进来。
> - TypeScript 也完全对得上：`useRulesRequired` 的类型是`(rules) => (key: string) => boolean` ，`isRequiredByRules` 的推断结果就是`(key: string) => boolean` ，所以调用时传`key` 合法。
>
> 因为「规则集合」在一张表里是 共享的 （所有列都用同一个`props.required` ），而「key」是 每列独有的 （`col.prop` 各不相同）。柯里化把这两者分离：
>
> 等价于把`rules` 提前「绑定」好，得到一个只差`key` 的函数，从而可以在`v-for` 中对所有列复用同一个判定逻辑。
>
> 如果不想用柯里化，也可以写成一次性收两个参数的形式
>
> 只是那样每次调用都要重复传`props.required` ，柯里化更省一点

### 说明

一句话： DFS 就是递归时先深入子节点、再处理兄弟节点的遍历方式 。

##### DFS = Depth-First Search（深度优先搜索） ，是一种遍历树/图的算法。

核心思想： 一条路走到黑，走不通再回头 。遇到一个节点，先往它的子节点钻，直到没有子节点了，再回退到上一层试别的分支。

遍历顺序： A → B → B1 → B2 → C → C1

##### BFS（广度优先搜索）

A → B → C → B1 → B2 → C1

#### 树的术语

```
              A              ← 有子节点（B、C），不是叶子
             / \
            B   C            ← B 有子节点（D、E），不是叶子；C 没有子节点 → 叶子
           / \
          D   E              ← D、E 没有子节点 → 叶子
```



| 术语                | 含义                        | 例子                         |
| ------------------- | --------------------------- | ---------------------------- |
| 子节点（children）  | 直接下一层                  | A 的子 = B、C；B 的子 = D、E |
| 后代（descendants） | 所有下层层级（子 + 孙 + …） | A 的后代 = B、C、D、E        |
| 叶子（leaf）        | 没有子节点 的节点           | C、D、E（它们下面没有了）    |
| 中间节点 / 分组     | 有子节点的节点              | A、B                         |
