```vue
  const { proxy } = getCurrentInstance()!
  function toggleSearch(): void {
    let el: HTMLElement | null = proxy!.$el
    let formEl: HTMLElement | null = null
    while ((el = el!.parentElement) && el !== document.body) {
      if ((formEl = el.querySelector('.el-form'))) break

    if (!formEl) return emits('update:showSearch', !props.showSearch)

  function animateSearch(el: HTMLElement, isHide: boolean): void {
    const DURATION = 260
    const TRANSITION = 'max-height 0.25s ease, opacity 0.2s ease'
    const clear = () => Object.assign(el.style, { transition: '', maxHeight: '', opacity: '', overflow: '' })
    Object.assign(el.style, { overflow: 'hidden', transition: '' })
    if (isHide) {
      Object.assign(el.style, { maxHeight: el.scrollHeight + 'px', opacity: '1', transition: TRANSITION })

      nextTick(() => {
        Object.assign(el.style, { maxHeight: '0', opacity: '0' })
        requestAnimationFrame(() => requestAnimationFrame(() => {
          Object.assign(el.style, { transition: TRANSITION, maxHeight: el.scrollHeight + 'px', opacity: '1' })
        }))
        setTimeout(clear, DURATION)
      })
    }

        const key = props.columns[item].key

      }

      if (saved && typeof saved === 'object') {
        if (Array.isArray(props.columns)) {
          props.columns.forEach((col: TableShowColumns, index: number) => {
            if (saved[index] !== undefined) col.visible = saved[index]
          })
        } else {
          Object.keys(props.columns).forEach(key => {
            if (saved[key] !== undefined) props.columns[key].visible = saved[key]
          })
        }
      }
    } catch (e) {}
  }
  if (props.showColumnsType == "transfer") {
    // transfer穿梭显隐列初始默认隐藏列
    if (Array.isArray(props.columns)) {
      for (let item in props.columns) {
        if (props.columns[item].visible === false) {
          value.value.push(parseInt(item))
        }
      }
    } else {
      Object.keys(props.columns).forEach((key, index) => {
        if (props.columns[key].visible === false) {
          value.value.push(index)
        }
      })
    }
  }

  // 单勾选
  function checkboxChange(event: any, key: string | number): void {
    if (Array.isArray(props.columns)) {
      const col = props.columns.filter((item: any) => item.key == key)[0]
      if (col) {
        col.visible = event
      }
    } else {
      ;(props.columns as any)[key].visible = event
    }
    saveStorage()
  }

  // 切换全选/反选
  function toggleCheckAll(): void {
    const newValue = !isChecked.value
    if (Array.isArray(props.columns)) {
      props.columns.forEach((col: TableShowColumns) => (col.visible = newValue))
    } else {
      Object.values(props.columns).forEach((col) => ((col as TableShowColumns).visible = newValue))
    }
    saveStorage()
  }

  // 将当前列显隐状态持久化到 localStorage
  function saveStorage(): void {
    if (!props.storageKey) return
    try {
      let state: { [key: string]: boolean } = {}
      if (Array.isArray(props.columns)) {
        props.columns.forEach((col: TableShowColumns, index: number) => { state[index] = col.visible })
      } else {
        Object.keys(props.columns).forEach(key => { state[key] = props.columns[key].visible })
      }
      cache.local.setJSON(props.storageKey, state)
    } catch (e) {}
  }
</script>
```

```ts
import type { Component } from "vue";

/** 内置按钮语义，覆盖项目里最常见的工具栏操作 */
export type ToolbarActionKey =
    | "add"
    | "edit"
    | "del"
    | "batch"
    | "import"
    | "export"
    | "download"
    | string;

/** 单个工具栏按钮配置 */
export interface ToolbarButton {
    /** 动作标识，点击后随 action 事件回传（如 'add' / 'export'），也用于匹配内置默认样式 */
    key: ToolbarActionKey;
    /** 按钮文案；不传则取内置默认（如 add -> "新增"） */
    label?: string;
    /** el-button 的 type；不传则取内置默认 */
    type?: "primary" | "success" | "warning" | "danger" | "info" | "";
    /** 按钮图标名（Element Plus 图标组件名，如 'Plus'）或图标组件；不传则取内置默认 */
    icon?: string | Component;
    /** 是否 plain 风格，默认 true（对齐项目现状） */
    plain?: boolean;
    /** 按钮尺寸，默认 'default' */
    size?: "large" | "default" | "small";
    /** 是否禁用 */
    disabled?: boolean;
    /** 是否隐藏该按钮（用于按权限/状态动态控制） */
    hidden?: boolean;
    /** 权限标识，透传给 v-hasPermi 指令（旧范式常用），为空则不加限制 */
    perm?: string;
}

export interface TableToolbarProps {
```

```vue
  import type { DefineListEmits } from '@/hooks/useList'
  import type { CompTableProp, CompTableSlots } from './types'

  import { useTempRefs } from '@/hooks/useTempRefs'
  import { createDynamicColumns } from './renderColumns'
  import { useColumnVisibility } from './useColumnVisibility'

  import TableToolbar from '@/components/TableToolbar/index.vue'

  const props = withDefaults(defineProps<CompTableProp<T, R>>(), {
    selected: () => [],
    keys: '',
    buttons: () => [],
    storageKey: '',
    showTableToolBar: true,
    showSearch: true,
    selectable: true,
    selectionWidth: 55,
    minWidth: 100,
    showSummary: false,
    summaryMethod: undefined,
    showRightToolbar: true,
    maxHeight: '',
    height: '',
    isAutoHeight: false,
    isFixedHeight: true,
    index: false,
    showToolBarSearch: true,
    required: false,
    bottomOffset: 20 + 20 + 32,
  })

  interface Emits
    extends Omit<DefineListEmits<T, R>, 'update:list'>,
      DefineCheckAllEmits<R> {
    /** 刷新按钮点击 */
    queryTable: []
    /** 搜索区显隐切换 */
    'update:showSearch': [val: boolean]
    /** List 模式回传整个 List；数组模式回传 items 数组 */
    'update:list': [value: List<T, R> | R[]]
  }
  const emit = defineEmits<Emits>()
  defineSlots<CompTableSlots<R>>()

  /** showSearch 双向绑定：TableToolbar 切换 → 透传给父组件 */
  const innerShowSearch = computed({
    get: () => props.showSearch,
    set: (val: boolean) => emit('update:showSearch', val),
  })

  // ---------- 表格高度 ----------
  const { ELTable } = useTempRefs('ELTable')
  const { tableHeight, refresh: refreshTableHeight } = useTableHeight(
    ELTable,
    props.bottomOffset
  )

  /**
   * 高度解析（优先级：isAutoHeight > height > isFixedHeight > maxHeight）
   * - isAutoHeight：忽略 height/maxHeight，表格自然撑开
   * - height：显式固定高度（不参与视口计算，避免滚动时高度递增）
   * - isFixedHeight：height=tableHeight（按视口剩余空间自动计算）
   * - maxHeight：兜底最大高度
   */
  const resolvedHeight = computed<number | string | undefined>(() => {
    if (props.isAutoHeight) return undefined
    if (props.height !== '') return props.height
    if (props.isFixedHeight) return tableHeight.value
    return undefined
  })
  const resolvedMaxHeight = computed<number | string | undefined>(() => {
    if (props.isAutoHeight) return undefined
    if (props.height !== '') return undefined
    if (props.isFixedHeight) return undefined
    return props.maxHeight || tableHeight.value
  })

  /** 搜索区显隐切换后表格 top 变化，需重新计算高度 */
  watch(
    () => props.showSearch,
    () => requestAnimationFrame(refreshTableHeight)
  )

  // ---------- 列表数据 ----------
  const adaptorList = useListAdaptor<R>(() => (Array.isArray(props.list) ? props.list : []))

  const list = computed<List<T, R>>({
    get: () =>
      Array.isArray(props.list) ? ((adaptorList as unknown) as List<T, R>) : props.list,
    set: val => emit('update:list', val),
  })

  watch(
    () => adaptorList.items,
    items => {
      if (Array.isArray(props.list) && toRaw(props.list) !== toRaw(items)) {
        emit('update:list', items as R[])
      }
    },
    { deep: true }
  )

  /** 跨页选择 v-model:selected + 全选 + 单行勾选 */
  const { isCheckAll, handleChecked } = useCheckAll(props.keys, list, props, { emit })

  // ---------- 列显隐（多级表头父子级联 / 持久化 / 渲染过滤） ----------
  const { columnVisibility, dataColumns, indexColumnConfig } = useColumnVisibility<R>({
    getColumns: () => props.columns,
    getStorageKey: () => props.storageKey,
  })

  // ---------- 动态数据列（递归渲染，详见 ./renderColumns） ----------
  const DynamicColumns = createDynamicColumns<R>({
    dataColumns,
    getMinWidth: () => props.minWidth,
    getRequired: () => props.required,
    getAlign: () => props.align,
    getHeaderAlign: () => props.headerAlign,
  })
</script>
```

```ts
import type { ComputedRef, VNode } from 'vue'
import type { CellScope, ColumnSlotProps, CompTableProp, HeaderScope, TableColumn } from './types'
import type { FormatRow } from '@/hooks/useFormatValue'

import { ElTableColumn } from 'element-plus'

import { useFormatValue } from '@/hooks/useFormatValue'
import { useRulesRequired } from '@/hooks/useRules'
import { RequiredHeader } from '@/hooks/useTable'

import CompPrice from '@/components/Price/index.vue'

interface CreateDynamicColumnsOptions<R extends Record<string, any>> {
  /** 已过滤的渲染列（分组 + 叶子） */
  dataColumns: ComputedRef<TableColumn<R>[]>
  /** 叶子列最小宽度（叶子未配置时兜底） */
  getMinWidth: () => CompTableProp<Record<string, any>, R>['minWidth']
  /** 表头必填红星规则（boolean 或 useRules 规则对象） */
  getRequired: () => CompTableProp<Record<string, any>, R>['required']
  /** 全局单元格对齐（列 col.align 优先） */
  getAlign: () => CompTableProp<Record<string, any>, R>['align']
  /** 全局表头对齐（列 col.headerAlign 优先，未传则跟随 align） */
  getHeaderAlign: () => CompTableProp<Record<string, any>, R>['headerAlign']
}

/** 渲染 vnode 用的兄弟级唯一 key */
const resolveColKey = <R>(col: TableColumn<R>, idx: number): string => col.key ?? String(col.prop) ?? String(idx)

/**
 * 创建动态数据列函数式组件：
 * - 支持 children 递归渲染多级表头
 * - 叶子单元格优先级：slot > price > format
 * - 表头支持 renderHeader 函数 / renderHeader=true 必填红星 / headerSlot 插槽
 *
 * 必须在 setup 同步执行期间调用（内部使用 useSlots）
 */
export const createDynamicColumns = <R extends Record<string, any>>(options: CreateDynamicColumnsOptions<R>) => {
  const { dataColumns, getMinWidth, getRequired, getAlign, getHeaderAlign } = options
  const slots = useSlots()
  const isRequiredByRules = useRulesRequired(getRequired)

  /** render-header 仅处理函数类型；true / headerSlot 走 #header 插槽 */
  const getRenderHeader = (col: TableColumn<R>) => {
    if (col.headerSlot || col.renderHeader === true) return undefined
    if (typeof col.renderHeader === 'function') return col.renderHeader
    return undefined
  }

  /** 调用 col-{prop} 插槽（单元格 / 表头共用，靠 header 参数区分） */
  const callColumnSlot = (col: TableColumn<R>, payload: Omit<ColumnSlotProps<R>, 'col'>): unknown =>
    slots[`col-${String(col.prop)}`]?.({ col, ...payload })

  /** 叶子列单元格内容（优先级：slot > price > format） */
  const renderLeafCell = (col: TableColumn<R>, scope: CellScope<R>) => {
    if (scope.$index < 0) return null
    if (col.slot) {
      return callColumnSlot(col, {
        header: false,
        row: scope.row,
        $index: scope.$index,
        column: scope.column,
        scope
      })
    }
    if (col.price) {
      return h(CompPrice, {
        price: scope.row[col.prop as keyof R],
        ...(col.priceConfig ?? {})
      })
    }
    return useFormatValue(scope.row[col.prop as keyof R] as FormatRow, col.format?.type as never, col.format?.formatStr as never)
  }

  /** 自定义表头：renderHeader=true 渲染必填红星；headerSlot 走 col-{prop} 插槽 */
  const renderColumnHeader = (col: TableColumn<R>, scope: HeaderScope<R>) => {
    if (col.renderHeader === true) {
      return h(RequiredHeader, {
        label: col.label,
        required: isRequiredByRules(String(col.prop))
      })
    }
    return callColumnSlot(col, {
      header: true,
      row: {} as R,
      $index: scope.$index,
      column: scope.column,
      scope
    })
  }

  /** 递归渲染单列：有 children 渲染为分组表头，否则为叶子数据列 */
  const renderColumn = (col: TableColumn<R>, idx: number, depth = 0): VNode => {
    const hasChildren = !!col.children?.length
    const columnSlots: Record<string, (scope: any) => unknown> = {}

    // 分组与叶子均支持自定义表头插槽 / 必填红星
    const useCustomHeader = col.headerSlot || col.renderHeader === true
    if (hasChildren) {
      columnSlots.default = () => col.children!.map((child, i) => renderColumn(child, i, depth + 1))
    } else {
      columnSlots.default = (scope: CellScope<R>) => renderLeafCell(col, scope)
    }
    if (useCustomHeader) {
      columnSlots.header = (scope: HeaderScope<R>) => renderColumnHeader(col, scope)
    }

    return h(
      ElTableColumn,
      {
        key: `${depth}-${resolveColKey(col, idx)}`,
        prop: hasChildren ? undefined : String(col.prop),
        label: col.label,
        width: col.width,
        minWidth: hasChildren ? col.minWidth : col.minWidth ?? getMinWidth(),
        fixed: col.fixed,
        align: col.align ?? getAlign(),
        headerAlign: col.headerAlign ?? getHeaderAlign() ?? col.align ?? getAlign(),
        showOverflowTooltip: col.showOverflowTooltip,
        renderHeader: getRenderHeader(col)
      },
      columnSlots
    )
  }

  /** 函数式组件：渲染期求值，自动追踪 dataColumns 显隐变化 */
  return () => dataColumns.value.map((col, idx) => renderColumn(col, idx))
}
```

#### `useColumnVisibility`

```ts
import type { ComputedRef } from 'vue'
import type { TableColumn } from './types'

import cache from '@/plugins/cache'

/** 显隐面板条目（RightToolbar 仅识别 label/visible，层级信息烤进 label） */
interface ColumnVisibilityItem {
  /** 列层级路径 key，全局唯一 */
  key: string
  /** 展示文案（子列带缩进前缀） */
  label: string
  /** 是否显示（RightToolbar 通过 v-model 直接修改） */
  visible: boolean
  /** 表头层级，0 为顶层 */
  level: number
  /** 父级路径 key，顶层列为空 */
  parentKey?: string
}

/** 全角空格不被 HTML 折叠，用于显隐面板中子列的层级缩进（RightToolbar 不支持层级结构） */
const LEVEL_INDENT = '　'

/** 列层级路径 key（全局唯一，显隐匹配 / vnode 渲染共用） */
export const resolveColPath = <R>(col: TableColumn<R>, parentPath = ''): string => `${parentPath}/${col.key ?? String(col.prop)}`

/**
 * 递归扁平化显隐列表（分组表头与子列均进入面板）：
 * - disabled 列不进入面板
 * - 分组列的子列全部 disabled 时，整组不进入面板
 * - 子列 label 前缀全角空格 + └ 表达层级
 */
const flattenVisibility = <R>(cols: TableColumn<R>[], level = 0, parentKey?: string, parentPath = ''): ColumnVisibilityItem[] =>
  cols.flatMap(col => {
    if (col.disabled) return []
    const key = resolveColPath(col, parentPath)
    const label = level > 0 ? `${LEVEL_INDENT.repeat(level)}└ ${col.label}` : col.label
    if (col.children?.length) {
      const children = flattenVisibility(col.children, level + 1, key, key)
      if (!children.length) return []
      return [{ key, label, visible: col.visible !== false, level, parentKey }, ...children]
    }

    return [{ key, label, visible: col.visible !== false, level, parentKey }]
  })

/** 查找首个匹配的叶子列 */
const findLeafColumn = <R>(cols: TableColumn<R>[], predicate: (col: TableColumn<R>) => boolean): TableColumn<R> | undefined => {
  for (const col of cols) {
    if (col.children?.length) {
      const hit = findLeafColumn(col.children, predicate)
      if (hit) return hit
    } else if (predicate(col)) {
      return col
    }
  }
  return undefined
}

interface UseColumnVisibilityOptions<R extends Record<string, any>> {
  /** 列配置（getter，保持对 props.columns 替换的响应） */
  getColumns: () => TableColumn<R>[]
  /** 显隐持久化 localStorage key，空串不持久化 */
  getStorageKey: () => string
}

interface UseColumnVisibilityReturn<R extends Record<string, any>> {
  /** 传给工具栏的扁平显隐列表（RightToolbar 直接改 visible） */
  columnVisibility: ColumnVisibilityItem[]
  /** 实际渲染的数据列（已按祖先显隐级联过滤，剔除 disabled/序号列） */
  dataColumns: ComputedRef<TableColumn<R>[]>
  /** 序号列样式覆盖配置（任意层级中 prop='index' 的叶子列） */
  indexColumnConfig: ComputedRef<TableColumn<R> | undefined>
}

/**
 * 多级表头列显隐：父子级联
 */
export const useColumnVisibility = <R extends Record<string, any>>(options: UseColumnVisibilityOptions<R>): UseColumnVisibilityReturn<R> => {
  const { getColumns, getStorageKey } = options

  /** 列显隐结构静态 */
  const columnVisibility = reactive(flattenVisibility(getColumns()))

  /** key -> 自身可见状态（渲染过滤时沿层级路径匹配） */
  const visibilityMap = computed(() => new Map(columnVisibility.map(item => [item.key, item.visible])))

  // ---------- 多级表头父子级联的索引缓存 ----------
  /** `visibilityItemMap` —— 按 key 查条目 */
  const visibilityItemMap = new Map(columnVisibility.map(item => [item.key, item]))
  /** 按 key 查直接子条目 */
  const visibilityChildrenMap = (() => {
    const map = new Map<string, ColumnVisibilityItem[]>()
    columnVisibility.forEach(item => {
      if (!item.parentKey) return
      const list = map.get(item.parentKey) ?? []
      list.push(item)
      map.set(item.parentKey, list)
    })
    return map
  })()

  /** 分组下全部后代 */
  const getDescendants = (key: string): ColumnVisibilityItem[] =>
    (visibilityChildrenMap.get(key) ?? []).flatMap(kid => [kid, ...getDescendants(kid.key)])

  /** 分组下全部叶子后代 */
  const getLeafDescendants = (key: string): ColumnVisibilityItem[] => getDescendants(key).filter(d => !visibilityChildrenMap.get(d.key)?.length)

  /**
   * 程序性级联标记
   */
  // 记录本批被程序级联修改的列 key。
  const cascadedKeys = new Set<string>()
  // 级联赋值的唯一入口：
  const setCascadedVisible = (item: ColumnVisibilityItem, visible: boolean) => {
    if (item.visible === visible) return
    cascadedKeys.add(item.key)
    item.visible = visible
  }

  /** 用户手动切换（RightToolbar 直接改 visible，无标记即视为用户操作） */
  const handleUserToggle = (item: ColumnVisibilityItem) => {
    if (visibilityChildrenMap.get(item.key)?.length) {
      // 分组勾选/取消：整组显隐，中间分组与叶子全部同步
      getDescendants(item.key).forEach(d => setCascadedVisible(d, item.visible))
      return
    }
    // 叶子切换：自底向上归并祖先，分组 visible = 子树是否存在可见叶子
    let parentKey = item.parentKey
    while (parentKey) {
      const parent = visibilityItemMap.get(parentKey)
      if (parent) {
        const anyVisible = getLeafDescendants(parent.key).some(leaf => leaf.visible)
        setCascadedVisible(parent, anyVisible)
      }
      parentKey = parent?.parentKey
    }
  }

  /** 级联完成后覆写 localStorage（RightToolbar 点击当刻写入的是级联前中间态） */
  const persistColumnVisibility = () => {
    if (!getStorageKey()) return
    const state: Record<number, boolean> = {}
    columnVisibility.forEach((item, index) => (state[index] = item.visible))
    cache.local.setJSON(getStorageKey(), state)
  }

  /**
   * 初始化保护：RightToolbar 挂载时会按 localStorage 恢复 visible，这批变更不是
   * 用户操作，watcher 一律忽略；挂载结束后按叶子状态自底向上归并分组显隐，
   * 修复历史存储中“父隐藏/子显示”的不一致
   */
  let visibilityReady = false
  onMounted(() =>
    nextTick(() => {
      ;[...columnVisibility].reverse().forEach(item => {
        if (visibilityChildrenMap.get(item.key)?.length) {
          item.visible = getLeafDescendants(item.key).some(leaf => leaf.visible)
        }
      })
      visibilityReady = true
      persistColumnVisibility()
    })
  )

  columnVisibility.forEach(item =>
    watch(
      () => item.visible,
      () => {
        if (!visibilityReady) return
        if (cascadedKeys.delete(item.key)) return
        handleUserToggle(item)
        persistColumnVisibility()
      }
    )
  )

  // ---------- 渲染过滤 ----------
  /**
   * 递归过滤多级表头列（渲染用）：
   * - 叶子生效可见 = 自身可见 && 所有祖先可见（父隐藏级联隐藏全部子列）
   * - disabled / prop='index' 叶子剔除（序号列单独静态渲染）
   * - 分组列无生效子列时整组剔除
   */
  const filterTreeColumns = (cols: TableColumn<R>[], parentPath = '', parentVisible = true): TableColumn<R>[] =>
    cols.flatMap(col => {
      if (col.disabled) return []
      const key = resolveColPath(col, parentPath)
      const ownVisible = visibilityMap.value.get(key) ?? col.visible !== false
      const effectiveVisible = parentVisible && ownVisible
      if (col.children?.length) {
        const children = filterTreeColumns(col.children, key, effectiveVisible)
        return children.length ? [{ ...col, children }] : []
      }
      if (col.prop === 'index') return []
      return effectiveVisible ? [col] : []
    })

  const dataColumns = computed(() => filterTreeColumns(getColumns()))

  const indexColumnConfig = computed(() => findLeafColumn(getColumns(), col => col.prop === 'index'))

  return { columnVisibility, dataColumns, indexColumnConfig }
}
```

> `const isRequiredByRules = useRulesRequired(() => props.required)`
>
> 高阶函数（闭包 / 柯里化）
>
> ```ts
> // 第 1 步：传入 rules，返回一个函数（此时还没判定任何东西）
> const isRequiredByRules = useRulesRequired(() => props.required)
> //    ↑ 此时 isRequiredByRules 的类型是 (key: string) => boolean，不是 boolean
> 
> // 第 2 步：调用这个返回的函数，传入 key，才真正执行判定
> isRequiredByRules('pkAccountId')  // → boolean
> ```
>
> 关键点：
>
> - useRulesRequired外层函数 接收`rules` ， 内层函数 接收`key` 。两者参数不同，所以不冲突、不报错。
> - `rules` 被内层函数通过 闭包 捕获，不需要第二次传进来。
> - TypeScript 也完全对得上：`useRulesRequired` 的类型是`(rules) => (key: string) => boolean` ，`isRequiredByRules` 的推断结果就是`(key: string) => boolean` ，所以调用时传`key` 合法。
>
> 因为「规则集合」在一张表里是 共享的 （所有列都用同一个`props.required` ），而「key」是 每列独有的 （`col.prop` 各不相同）。柯里化把这两者分离：
>
> 等价于把`rules` 提前「绑定」好，得到一个只差`key` 的函数，从而可以在`v-for` 中对所有列复用同一个判定逻辑。
>
> 如果不想用柯里化，也可以写成一次性收两个参数的形式
>
> 只是那样每次调用都要重复传`props.required` ，柯里化更省一点
