### price组件（展示）

```vue
<template>
  <span class="price">
    <!--type start-->
    <span
      v-if="type"
      class="type">
      {{ type }}
    </span>
    <!--type end-->

    <!--sign start（负数时符号前置，如 "-￥5.00"）-->
    <span
      v-if="pix < 0 && showSign"
      class="sign">
      -
    </span>
    <span
      v-if="showSign && !isNaN(pix)"
      class="sign"
      :style="signStyle">
      ￥
    </span>
    <!--sign end-->

    <!--number start-->
    <span class="number">
      <span>{{ int }}</span>
      <span
        v-if="decimal"
        class="decimal">
        {{ decimal }}
      </span>
    </span>
    <span
      v-if="unit"
      class="unit">
      {{ unit }}
    </span>
    <!--number end-->
  </span>
</template>

<script lang="ts">
  export default {
    name: 'Price'
  }
</script>

<script lang="ts" setup>
  import { ToThousandFilter } from '@/filters'

  export interface Props {
    /** 是否显示符号 */
    showSign?: boolean
    /** 价格（null/undefined/空字符串 视为空值，展示为 --） */
    price?: string | number | null
    /** 前置符号 */
    type?: string
    /** 是否显示小数部分 */
    double?: boolean
    /** 后置单位 */
    unit?: string
    /** 价格单位 */
    priceUnit?: string
    /** 符号样式 */
    signStyle?: string
    /** 是否使用科技计数法 */
    isThousand?: boolean
  }

  const props = withDefaults(defineProps<Props>(), {
    showSign: true,
    price: '',
    type: '',
    double: false,
    unit: '',
    priceUnit: 'fen',
    signStyle: '',
    isThousand: false
  })

  /** 是否为空值 */
  const isEmpty = computed(() => props.price === '' || props.price === null || props.price === undefined)

  /** 价格（空值或非法数字时为 NaN） */
  const pix = computed(() => {
    if (isEmpty.value) return NaN
    return props.priceUnit === 'fen' ? Number(props.price) / 100 : Number(props.price)
  })

  /** 整数部分 */
  const int = computed(() => {
    if (isNaN(pix.value)) return '--'
    const absPix = Math.abs(pix.value)
    const intStr = absPix.toFixed(2).split('.')[0]
    return props.isThousand ? ToThousandFilter(intStr) : intStr
  })

  /** 小数部分 */
  const decimal = computed(() => {
    if (isNaN(pix.value)) return ''
    const absPix = Math.abs(pix.value)
    let num: string = absPix.toFixed(2).split('.')[1]
    if (!props.double) {
      if (Number(num) === 0) return ''
      if (Math.floor(Number(num) / 10) === Number(num) / 10) num = (Number(num) / 10).toString()
    }
    return '.' + num
  })
</script>

<style lang="scss" scoped>
  .price {
    display: inline-flex;
    align-items: baseline;
    justify-content: flex-start;
  }
</style>
```

### Input-price（输入框）

```vue
<template>
  <el-input
    v-model="displayValue"
    type="number"
    :disabled="disabled"
    clearable
    :size="size"
    :step="0.01"
    :min="min"
    :max="max"
    @blur="handleCommit"
    @change="handleCommit"
    @keyup.enter="handleCommit">
    <template
      v-if="slots.prefix"
      #prefix>
      <slot name="prefix" />
    </template>
    <template
      v-if="slots.suffix"
      #suffix>
      <slot name="suffix" />
    </template>
    <template
      v-if="slots.prepend"
      #prepend>
      <slot name="prepend" />
    </template>
    <template
      v-if="slots.append"
      #append>
      <slot name="append" />
    </template>
  </el-input>
</template>

<script lang="ts">
  export default {
    name: 'InputPrice'
  }
</script>

<script lang="ts" setup>
  import { useDivide, useMulti } from '@/hooks/useCalc'

  /** 实际值 */
  const modelValue = defineModel<number | string>({ default: '' })

  interface Props {
    /** 分↔元转换比例，默认 100（1元=100分） */
    ratio?: number
    /** 尺寸 */
    size?: '' | 'default' | 'small' | 'large'
    /** 允许输入的最大值（元单位） */
    max?: number
    /** 允许输入的最小值（元单位） */
    min?: number
    /** 禁用 */
    disabled?: boolean
  }
  const props = withDefaults(defineProps<Props>(), {
    ratio: 100,
    size: 'default',
    max: Infinity,
    min: 0,
    disabled: false,
  })

  const slots = useSlots()

  /** 显示值 */
  const displayValue = ref<string>('')

  watch(
    modelValue,
    val => {
      if (val === '' || val === null || val === undefined || isNaN(Number(val))) {
        displayValue.value = ''
      } else {
        displayValue.value = String(useDivide(Number(val), props.ratio))
      }
    },
    { immediate: true }
  )

  const handleCommit = () => {
    const raw = displayValue.value.trim()
    if (raw === '' || raw === '-' || raw === '.' || raw === '-.') {
      modelValue.value = ''
      return
    }
    const num = Number(raw)
    if (isNaN(num)) {
      // 非法输入，回退到上一次有效值
      modelValue.value = modelValue.value
      return
    }
    const clamped = Math.min(Math.max(num, props.min), props.max)
    displayValue.value = String(clamped)
    modelValue.value = useMulti(clamped, props.ratio)
  }
</script>

<style lang="scss">
  /* 隐藏 el-input type=number 的原生上下箭头 */
  .el-input__inner {
    -moz-appearance: textfield;

    &::-webkit-outer-spin-button,
    &::-webkit-inner-spin-button {
      -webkit-appearance: none;
      margin: 0;
    }
  }
</style>

```
