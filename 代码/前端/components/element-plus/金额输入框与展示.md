price组件（展示）

```vue
<template>
  <span class="price">
    <!--type start-->
    <span
      v-if="type"
      class="type">
      {{ type }}
    </span>
    <!--type end-->

    <!--sign start-->
    <span
      v-if="showSign && !isNaN(pix)"
      class="sign"
      :style="signStyle">
      ￥
    </span>
    <!--sign end-->

    <!--number start-->
    <span class="number">
      <span>{{ int }}</span>
      <span
        v-if="decimal"
        class="decimal">
        {{ decimal }}
      </span>
    </span>
    <span
      v-if="unit"
      class="unit">
      {{ unit }}
    </span>
    <!--number end-->
  </span>
</template>

<script lang="ts">
  export default {
    name: 'Price'
  }
</script>

<script lang="ts" setup>
  // import { computed } from 'vue'

  import { ToThousandFilter } from '@/filters'

  export interface Props {
    /** 是否显示符号 */
    showSign?: boolean
    /** 价格（null/undefined/空字符串 视为空值，展示为 --） */
    price?: string | number | null
    /** 前置符号 */
    type?: string
    /** 是否显示小数部分 */
    double?: boolean
    /** 后置单位 */
    unit?: string
    /** 价格单位 */
    priceUnit?: string
    /** 符号样式 */
    signStyle?: string
    /** 是否使用科技计数法 */
    isThousand?: boolean
  }

  const props = withDefaults(defineProps<Props>(), {
    showSign: true,
    price: '',
    type: '',
    double: false,
    unit: '',
    priceUnit: 'fen',
    signStyle: '',
    isThousand: false
  })

  /** 是否为空值 */
  const isEmpty = computed(() => props.price === '' || props.price === null || props.price === undefined)

  /** 价格（空值或非法数字时为 NaN） */
  const pix = computed(() => {
    if (isEmpty.value) return NaN
    return props.priceUnit === 'fen' ? Number(props.price) / 100 : Number(props.price)
  })

  /** 整数部分 */
  const int = computed(() => {
    if (isNaN(pix.value)) return '--'
    const int = pix.value.toFixed(2).split('.')[0]
    return props.isThousand ? ToThousandFilter(int) : int
  })

  /** 小数部分 */
  const decimal = computed(() => {
    if (isNaN(pix.value)) return ''
    let num: string = pix.value.toFixed(2).split('.')[1]
    if (!props.double) {
      if (Number(num) === 0) return ''
      if (Math.floor(Number(num) / 10) === Number(num) / 10) num = (Number(num) / 10).toString()
    }
    return '.' + num
  })
</script>

<style lang="scss" scoped>
  .price {
    display: inline-flex;
    align-items: baseline;
    justify-content: flex-start;
  }
</style>
```

Input-price（输入框）

```vue
<template>
  <el-input
    v-model.trim="value"
    type="number"
    :disabled="disabled"
    clearable
    :size="size"
    :step="0.01"
    :min="0">
    <template
      v-if="slots.prefix"
      #prefix>
      <slot name="prefix" />
    </template>
    <template
      v-if="slots.suffix"
      #suffix>
      <slot name="suffix" />
    </template>
    <template
      v-if="slots.prepend"
      #prepend>
      <slot name="prepend" />
    </template>
    <template
      v-if="slots.append"
      #append>
      <slot name="append" />
    </template>
  </el-input>
</template>

<script lang="ts">
  export default {
    name: 'InputPrice'
  }
</script>

<script lang="ts" setup>
  // import type { PropType } from 'vue'
  // import { onBeforeMount, ref, useSlots, watch } from 'vue'

  import { useDivide, useMulti } from '@/hooks/useCalc'
  import { defineModelValueProps } from '@/hooks/useModelValue'

  const props = defineProps({
    /** 定义model值参数 */
    ...defineModelValueProps<number | string>([String, Number]),
    /** 比例 */
    ratio: {
      type: Number,
      default: 100
    },
    /** 尺寸 */
    size: {
      type: String as PropType<'' | 'default' | 'small' | 'large'>,
      default: 'large'
    },
    /** 限制可以输入的最大值 */
    max: {
      type: Number as PropType<number>,
      default: Infinity
    },
    /** 限制可以输入的最小值 */
    min: {
      type: Number as PropType<number>,
      default: 0
    },
    /** 禁用 */
    disabled: {
      type: Boolean,
      default: false
    }
  })
  const emit = defineEmits(['update:modelValue'])
  const slots = useSlots()

  /** value */
  const value = ref(props.modelValue)

  onBeforeMount(() => {
    if (props.modelValue === '' || isNaN(Number(props.modelValue))) value.value = ''
    else value.value = useDivide(Number(props.modelValue), props.ratio)
  })

  watch(
    () => props.modelValue,
    () => {
      if (props.modelValue === '' || isNaN(Number(props.modelValue))) value.value = ''
      else {
        const price = useDivide(Number(props.modelValue), props.ratio)
        if (price > props.max) value.value = props.max
        else if (price < props.min) value.value = props.min
        else value.value = price
      }
    }
  )

  watch(
    () => value.value,
    () => {
      if (value.value === '') emit('update:modelValue', value.value)
      else if (!isNaN(Number(value.value))) {
        const number = useMulti(Number(value.value), props.ratio)
        emit('update:modelValue', number)
      } else emit('update:modelValue', '')
    }
  )
</script>

<style lang="scss">
  /* 隐藏 el-input type=number 的原生上下箭头 */
  .el-input__inner {
    -moz-appearance: textfield;

    &::-webkit-outer-spin-button,
    &::-webkit-inner-spin-button {
      -webkit-appearance: none;
      margin: 0;
    }
  }
</style>
```
