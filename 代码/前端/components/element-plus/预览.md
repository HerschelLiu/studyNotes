`Preview`

```vue
<template>
  <CompDrawer
    v-model="show"
    title="预览"
    :enable-scroll="false"
    :show-save-button="false">
    <!-- Word 预览（.docx） -->
    <div
      v-if="previewType === 'doc'"
      v-loading="isOfficeLoading"
      class="office-preview">
      <vue-office-docx
        v-if="previewUrl"
        :src="previewUrl" />
    </div>
    <!-- Excel 预览（.xlsx） -->
    <div
      v-else-if="previewType === 'xls'"
      v-loading="isOfficeLoading"
      class="office-preview">
      <vue-office-excel
        v-if="previewUrl"
        :src="previewUrl" />
    </div>
    <!-- 图片预览（.jpg/.png/.gif/.webp/.bmp/.svg 等） -->
    <div
      v-else-if="previewType === 'image'"
      v-loading="isIframeLoading"
      class="office-preview image-preview">
      <ImagePreview
        v-if="previewUrl"
        :src="previewUrl as string"
        alt="预览图片" />
    </div>
    <!-- 不支持在线预览的格式（.doc/.xls/.ppt/.pptx 等） -->
    <div
      v-else-if="previewType === 'unsupported'"
      class="office-preview">
      <el-empty :description="`暂不支持 .${ext} 格式在线预览，请下载后查看`" />
    </div>
    <!-- 图片 / PDF / 文本 / 视频等浏览器可渲染的格式 -->
    <IFrame
      v-else
      v-loading="isIframeLoading"
      :src="previewUrl as string" />
  </CompDrawer>
</template>

<script lang="ts" setup>
  import { readProfileUpload } from '@/api/basic/common'
  import { commonDownloadResource } from '@/api/common'
  import VueOfficeDocx from '@vue-office/docx'
  import '@vue-office/docx/lib/index.css'
  import VueOfficeExcel from '@vue-office/excel'
  import '@vue-office/excel/lib/index.css'

  import CompDrawer from '@/components/Drawer/index.vue'
  import IFrame from '@/components/iFrame/index.vue'
  import ImagePreview from '@/components/ImagePreview/index.vue'

  interface Props {
    url: string
    minIO?: boolean
  }
  const props = withDefaults(defineProps<Props>(), {
    minIO: true,
  })

  const emit = defineEmits([])

  const show = defineModel<boolean>()

  const { data: officeUrl, isLoading: isOfficeLoading } = useQuery({
    queryKey: ['PREVIEW_URL'],
    queryFn: async () => {
      const res = await commonDownloadResource(props.url)
      return res
    },
    enabled: computed(() => show.value && !!props.url && props.minIO && (previewType.value === 'doc' || previewType.value === 'xls')),
  })

  const { data: iframeUrl, isLoading: isIframeLoading } = useQuery({
    queryKey: ['PREVIEW_URL'],
    queryFn: async () => {
      const { url } = await readProfileUpload(props.url)
      return url
    },
    enabled: computed(() => show.value && !!props.url && props.minIO && (previewType.value === 'image' || previewType.value === 'iframe')),
  })

  const ext = computed(() => {
    const path = props.url.split('?')[0]
    const idx = path.lastIndexOf('.')
    return idx > -1 ? path.slice(idx + 1).toLowerCase() : ''
  })

  const previewType = computed<'doc' | 'xls' | 'image' | 'unsupported' | 'iframe'>(() => {
    const e = ext.value
    const doc = ['docx', 'doc']
    const xls = ['xls', 'xlsx']
    const image = ['jpg', 'jpeg', 'png', 'gif', 'webp', 'bmp', 'svg']
    const ppt = ['ppt', 'pptx']

    if (doc.includes(e)) return 'doc'
    if (xls.includes(e)) return 'xls'
    if (image.includes(e)) return 'image'
    if (ppt.includes(e)) return 'unsupported'
    return 'iframe'
  })

  const previewUrl = computed(() => {
    return officeUrl.value || iframeUrl.value || props.url
  })
</script>

<style lang="scss" scoped>
  .office-preview {
    height: 60vh;
    overflow: auto;
  }

  .image-preview {
    display: flex;
    align-items: center;
    justify-content: center;

    img {
      max-width: 100%;
      max-height: 100%;
      object-fit: contain;
    }
  }
</style>
```

`api`

```ts
import type { AjaxResult } from '@/types'

import request from '@/utils/request'

/** 访问 MinIO 中的上传文件 */
export const readProfileUpload = (resource: string): Promise<{ url: string }> =>
  request({
    url: `/common/preview`,
    method: 'get',
    params: {
      resource
    },
    headers: {
      'response-content-disposition': 'inline'
    }
  })

import { getToken } from '@/utils/auth'

export const commonDownloadResource = (resource: string): Promise<ArrayBuffer> => {
  return request({
    url: '/common/download/resource',
    params: { resource },
    method: 'get',
    responseType: 'arraybuffer',
    headers: { Authorization: 'Bearer ' + getToken() }
  })
}
```
