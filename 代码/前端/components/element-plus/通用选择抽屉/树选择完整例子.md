```vue
<template>
  <comp-drawer
    v-model="show"
    :z-index="zIndex"
    :enable-scroll="false"
    title="归口科室"
    :tip="tip"
    @save="handleSave">
    <div class="app-container">
      <!-- 搜索条件 start -->
      <CompTableSearch
        :model="list.query"
        :loading="list.loading"
        @search="handleSearch"
        @reset="handleReset">
        <template #left>
          <CompCommonSearchItem
            v-model="list.query"
            :options="searchOptions" />
        </template>

        <template #right>
          <el-text class="mr10">{{ selectedRows.length }} / {{ maxLength }}</el-text>
          <el-button-group v-if="maxLength > 1">
            <el-button
              size="default"
              type="danger"
              @click="handleRemoveAll">
              清除所有
            </el-button>
            <el-button
              size="default"
              @click="handleSelectAll">
              全选
            </el-button>
          </el-button-group>
        </template>
      </CompTableSearch>
      <!-- 搜索条件 end -->

      <!-- 表格 start -->
      <CompTable
        :list="list"
        :columns="columns"
        :selectable="false"
        :show-table-tool-bar="false"
        index
        keys="id"
        default-expand-all>
        <template #col-action="{ row }">
          <ActionButton :row="row" />
        </template>
      </CompTable>
      <!-- 表格 end -->
    </div>
  </comp-drawer>
</template>

<script lang="ts" setup>
  import type { RequestListRollDept, ResponseListRollDept } from '@/api/masterBudget/budgetInfo'
  import type { CommonSearchItemOptions } from '@/components/CommonSearchItem/types'
  import type { TableColumn } from '@/components/CompTable/types'

  import { ElButton } from 'element-plus'
  import { treeRollDept } from '@/api/masterBudget/budgetInfo'
  import { YesOrNo } from '@/enum'
  import { useAssign } from '@/hooks/useObject'
  import { useError } from '@/hooks/useTip'

  import CompDrawer from '@/components/Drawer/index.vue'
  import CompTableSearch from '@/components/TableSearch/index.vue'
  import CompCommonSearchItem from '@/components/CommonSearchItem/index.vue'
  import CompTable from '@/components/CompTable/index.vue'

  export type RequestList = RequestListRollDept
  export type ResponseList = ResponseListRollDept

  interface Props {
    /** 描述标题 */
    tip?: string
    /** 指定弹窗层级 */
    zIndex?: number
    /** 已选择数据（可只传 id） */
    selected: Partial<ResponseList>[]
    /** 最小选择数据 */
    minLength?: number
    /** 最大选择数据，0 表示不限制 */
    maxLength?: number
    /** 禁用选择行 */
    disabledRow?: (row: Partial<ResponseList>) => boolean
    /** 是否只允许选择末级节点 */
    leafOnly?: boolean
    /**
     * 选中的父级是否连带其所有子级一起返回
     * - true：父级选中时，其下所有子级也在 selected 中
     * - false：只返回最上层被选中的节点
     */
    includeChildren?: boolean
  }

  const props = withDefaults(defineProps<Props>(), {
    zIndex: 1,
    selected: () => [],
    minLength: 1,
    maxLength: 500,
    disabledRow: () => false,
    leafOnly: false,
    includeChildren: false,
  })
  const emit = defineEmits(['change'])
  const show = defineModel<boolean>({
    default: false,
  })

  /** 客户端过滤：命中节点保留整棵子树，未命中但有命中后代的节点保留路径 */
  const filterTree = (nodes: ResponseList[], code: string, name: string): ResponseList[] => {
    const result: ResponseList[] = []
    nodes.forEach(node => {
      const children = filterTree(node.children ?? [], code, name)
      const isMatch =
        (!code || node.rollCode?.includes(code)) &&
        (!name || node.rollName?.includes(name))
      if (isMatch || children.length) result.push({ ...node, children })
    })
    return result
  }

  /** 全量树（选择/回显的真实数据源，不受过滤影响） */
  const fullTree = shallowRef<ResponseList[]>([])

  /** 树数据（树接口无分页，list.query 即搜索条件，queryFn 内客户端过滤） */
  const { list, getList } = useListQuery<RequestList, Partial<ResponseList>>(
    { rollCode: '', rollName: '' },
    () => queryKeys.rollDept.tree(),
    async query => {
      const res = await treeRollDept()
      const data = res.data ?? []
      fullTree.value = data
      const rollCode = (query.rollCode ?? '').trim()
      const rollName = (query.rollName ?? '').trim()
      return { ...res, data: filterTree(data, rollCode, rollName) }
    },
    { enablePage: false, enabled: computed(() => show.value) }
  )

  /** 节点索引（id → 节点 / 父节点，基于全量树） */
  const treeIndex = computed(() => {
    const nodeMap = new Map<string, ResponseList>()
    const parentMap = new Map<string, ResponseList>()
    const walk = (nodes: ResponseList[], parent?: ResponseList) => {
      nodes.forEach(node => {
        const id = String(node.id)
        nodeMap.set(id, node)
        if (parent) parentMap.set(id, parent)
        walk(node.children ?? [], node)
      })
    }
    walk(fullTree.value)
    return { nodeMap, parentMap }
  })

  /** 已勾选的节点 id（UI 唯一真相，不受过滤影响） */
  const checkedIds = ref<Set<string>>(new Set())

  /** 节点及其全部后代 id */
  const collectIds = (node: ResponseList): string[] => [
    String(node.id),
    ...(node.children ?? []).flatMap(collectIds),
  ]

  /** 判断末级节点 */
  const isLeafNode = (row: Partial<ResponseList>) => {
    const node = treeIndex.value.nodeMap.get(String(row.id)) ?? row
    return node.isLeaf === YesOrNo.是 || !node.children?.length
  }

  const isCheckedRow = (row: Partial<ResponseList>) => checkedIds.value.has(String(row.id))

  /** 子级全选时同步父级选中状态 */
  const syncParents = (set: Set<string>, node: ResponseList) => {
    let parent = treeIndex.value.parentMap.get(String(node.id))
    while (parent) {
      const children = parent.children ?? []
      const allChecked = children.length > 0 && children.every(child => set.has(String(child.id)))
      allChecked ? set.add(String(parent.id)) : set.delete(String(parent.id))
      parent = treeIndex.value.parentMap.get(String(parent.id))
    }
  }

  /** 选择/取消单个节点（非仅末级时父子联动） */
  const handleToggle = (row: Partial<ResponseList>) => {
    const node = treeIndex.value.nodeMap.get(String(row.id))
    if (!node) return
    const willCheck = !checkedIds.value.has(String(node.id))
    const next = new Set(checkedIds.value)
    if (props.leafOnly) {
      willCheck ? next.add(String(node.id)) : next.delete(String(node.id))
    } else {
      collectIds(node).forEach(id => (willCheck ? next.add(id) : next.delete(id)))
      syncParents(next, node)
    }
    if (isOverLimit(next)) {
      useError('选择数量已超过限制')
      return
    }
    checkedIds.value = next
  }

  /** 清除所有 */
  const handleRemoveAll = () => {
    checkedIds.value = new Set()
  }

  /** 按树顺序输出勾选结果 */
  const collectSelectedRows = (set: Set<string>): ResponseList[] => {
    const { parentMap } = treeIndex.value
    const rows: ResponseList[] = []
    const walk = (nodes: ResponseList[]) => {
      nodes.forEach(node => {
        const id = String(node.id)
        if (set.has(id)) {
          const parent = parentMap.get(id)
          if (props.includeChildren || !parent || !set.has(String(parent.id))) {
            rows.push(node)
          }
        }
        walk(node.children ?? [])
      })
    }
    walk(fullTree.value)
    return rows
  }

  const selectedRows = computed<ResponseList[]>(() => collectSelectedRows(checkedIds.value))

  /** 数量校验（以实际返回的 selected 为准） */
  const isOverLimit = (set: Set<string>) =>
    props.maxLength > 0 && collectSelectedRows(set).length > props.maxLength

  /** 全选 */
  const handleSelectAll = () => {
    const next = new Set(checkedIds.value)
    const walk = (nodes: ResponseList[]) => {
      nodes.forEach(node => {
        if (props.disabledRow(node)) return walk(node.children ?? [])
        if (props.leafOnly && !isLeafNode(node)) return walk(node.children ?? [])
        collectIds(node).forEach(id => next.add(id))
        walk(node.children ?? [])
      })
    }
    walk(fullTree.value)
    if (isOverLimit(next)) {
      useError('选择数量已超过限制')
      return
    }
    checkedIds.value = next
  }

  /** 回显 */
  const isEchoed = ref(false)
  const echoChecked = () => {
    const { nodeMap } = treeIndex.value
    const next = new Set<string>()
    props.selected.forEach(item => {
      const node = nodeMap.get(String(item.id))
      if (!node) return
      if (props.leafOnly) {
        if (isLeafNode(node)) next.add(String(node.id))
        return
      }
      collectIds(node).forEach(id => next.add(id))
      syncParents(next, node)
    })
    checkedIds.value = next
  }

  watch(
    [() => show.value, fullTree],
    ([visible]) => {
      if (!visible) {
        isEchoed.value = false
        return
      }
      if (isEchoed.value || !fullTree.value.length) return
      echoChecked()
      isEchoed.value = true
    },
    { immediate: true }
  )

  /** 操作按钮 */
  const renderAction = (row: Partial<ResponseList>) => {
    if (props.leafOnly && !isLeafNode(row)) return null
    const checked = isCheckedRow(row)
    return h(
      ElButton,
      {
        type: checked ? 'warning' : 'primary',
        size: 'default',
        link: true,
        disabled: props.disabledRow(row),
        onClick: () => handleToggle(row),
      },
      () => (checked ? '取消' : '选择')
    )
  }

  const ActionButton = (actionProps: { row: Partial<ResponseList> }) => renderAction(actionProps.row)

  /** 表格列配置 */
  const { budget_dept_data_source } = useDict('budget_dept_data_source')

  const columns: TableColumn<Partial<ResponseList>>[] = [
    { prop: 'rollCode', label: '科室编码' },
    { prop: 'rollName', label: '科室名称' },
    { prop: 'parentRollDeptName', label: '上级科室' },
    { prop: 'budgetUnitName', label: '预算单位' },
    { prop: 'isVirtual', label: '是否虚拟组织', format: { type: 'boolean' } },
    {
      prop: 'dataSource',
      label: '数据来源',
      format: { type: 'dict', formatStr: budget_dept_data_source },
    },
    { prop: 'action', label: '操作', width: 100, slot: true },
  ]

  /** 选择保存 */
  const handleSave = () => {
    const rows = selectedRows.value
    if (rows.length < props.minLength) {
      useError(`至少需要选择${props.minLength}条数据`)
      return
    }
    if (props.maxLength > 0 && rows.length > props.maxLength) {
      useError('选择数量已超过限制')
      return
    }
    emit('change', rows)
    show.value = false
  }

  /** 搜索项 */
  const searchOptions = computed<CommonSearchItemOptions<RequestList>[]>(() => [
    { prop: 'rollCode', label: '科室编码' },
    { prop: 'rollName', label: '科室名称' },
  ])

  /** 搜索 */
  const handleSearch = () => {
    getList()
  }

  /** 重置 */
  const handleReset = () => {
    useAssign(list.query, { rollCode: '', rollName: '' })
    getList()
  }
</script>
```
