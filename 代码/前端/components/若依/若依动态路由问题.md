在若依框架`系统管理-菜单管理`中为页面填写传参时，若填写的json不规范，会报错导致整个页面无法加载，为此增加修复

`src/layout/components/Sidebar/SidebarItem.vue`

```vue
<script lang="ts" setup>
/** 解析菜单 query；任何异常一律返回 null，不影响 sidebar 渲染 */
function parseMenuQuery(routeQuery: string): Record<string, any> {
  if (!routeQuery) return {}
  try {
    const parsed = JSON.parse(routeQuery)
    if (parsed !== null && typeof parsed === 'object') return parsed
  } catch {
    console.warn(`[SidebarItem] 菜单 query 不是合法 JSON，已忽略：${routeQuery}`)
  }
  return {}
}
function resolvePath(routePath: string, routeQuery?: string): string | { path: string; query: Record<string, any> } {
  if (isExternal(routePath)) {
    return routePath
  }
  if (isExternal(props.basePath)) {
    return props.basePath
  }
  if (routeQuery) {
    const query = routeQuery ? parseMenuQuery(routeQuery) : {} // !这行
    return { path: getNormalPath(props.basePath + '/' + routePath), query: query }
  }
  return getNormalPath(props.basePath + '/' + routePath)
}
</script>
```
