```vue
<template>
  <el-table
    v-loading="list.loading"
    :data="list.items"
    element-loading-text="正在加载..."
    style="width: 100%"
    border>
    <el-table-column
      label="选择"
      width="100"
      fixed>
      <template #header>
        <el-dropdown
          :show-timeout="0"
          placement="top"
          class="checked-dropdown">
          <el-checkbox v-model="isCheckAll">全选</el-checkbox>
          <template #dropdown>
            <el-dropdown-menu class="checked-dropdown-menu">
              <el-dropdown-item>
                <el-button
                  type="text"
                  size="default"
                  @click="cannelSelected">
                  取消所有
                </el-button>
              </el-dropdown-item>
            </el-dropdown-menu>
          </template>
        </el-dropdown>
      </template>

      <template #default="{ row }">
        <el-checkbox
          v-model="row.checked"
          @change="handleChecked(row)" />
      </template>
    </el-table-column>

    <el-table-column label="单位编码">
      <template #default="{ row }">{{ row.saleChannel || '--' }}</template>
    </el-table-column>
  </el-table>
</template>

<script lang="ts" setup>
  import { defineListEmits, defineListProps, useListRef } from '@/hooks/useList'
  import {
    defineCheckAllEmits,
    defineCheckAllProps,
    useCheckAll,
  } from '@/hooks/useCheckAll'

  const props = defineProps({
    /** 定义列表参数 */
    ...defineListProps(),
    /** 定义跨页选择参数 */
    ...defineCheckAllProps(),
  })

  const emit = defineEmits([
    /** 定义列表基础emits */
    ...defineListEmits(),
    /** 定义跨页选择基础emits */
    ...defineCheckAllEmits(),
  ])

  /** 列表 */
  const { list } = useListRef(props, { emit })

  /** 跨页选择 */
  const { isCheckAll, cannelSelected, handleChecked } = useCheckAll(
    props.keys,
    list,
    props,
    { emit }
  )
</script>
```
