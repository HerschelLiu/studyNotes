```vue
<template>
  <div class="app-container">
    <!-- 按钮组 start -->
    <el-row
      :gutter="10"
      class="mb8">
      <el-col :span="1.5">
        <el-button
          type="primary"
          size="small"
          @click="handleSave">
          保存
        </el-button>
      </el-col>

      <el-col :span="1.5">
        <el-button
          type="success"
          size="small"
          @click="handleAdd">
          新增
        </el-button>
      </el-col>
    </el-row>
    <!-- 按钮组 end -->

    <el-form
      ref="ELForm"
      label-position="right"
      :rules="rules"
      :model="form"
      :validate-on-rule-change="false"
      :disabled="state.loading"
      :inline="false"
      label-width="0">
      <el-form-item
        prop="items"
        label-width="0">
      	<el-table
        v-loading="state.loading"
        :data="form.items"
        element-loading-text="正在加载..."
        style="width: 100%"
        border>
        <el-table-column
          type="index"
          label="序号"
          width="60"
          align="center"
          fixed="left" />

        <el-table-column
          label="编码"
          min-width="200"
          :render-header="renderRequiredHeader('编码', 'code')">
          <template #default="{ row, $index }">
            <el-form-item
              :prop="`items.${$index}.code`"
              :rules="getCellRules('code')"
              label-width="0"
              class="table-form-item">
              <el-input
                v-model="row.code"
                placeholder="编码"
                clearable />
            </el-form-item>
          </template>
        </el-table-column>

        <el-table-column
          label="名称"
          min-width="200"
          :render-header="renderRequiredHeader('名称', 'name')">
          <template #default="{ row, $index }">
            <el-form-item
              :prop="`items.${$index}.name`"
              :rules="getCellRules('name')"
              label-width="0"
              class="table-form-item">
              <el-input
                v-model="row.name"
                placeholder="名称"
                clearable />
            </el-form-item>
          </template>
        </el-table-column>

        <el-table-column
          label="金额"
          width="150"
          :render-header="renderRequiredHeader('金额', 'amount')">
          <template #default="{ row, $index }">
            <el-form-item
              :prop="`items.${$index}.amount`"
              :rules="getCellRules('amount')"
              label-width="0"
              class="table-form-item">
              <el-input-number
                v-model="row.amount"
                :min="0"
                :precision="2"
                :step="100"
                controls-position="right"
                style="width: 100%" />
            </el-form-item>
          </template>
        </el-table-column>

        <el-table-column
          label="操作"
          width="100"
          fixed="right">
          <template #default="{ $index }">
            <el-button
              type="danger"
              text
              @click="handleDelete($index)">
              删除
            </el-button>
          </template>
        </el-table-column>
      </el-table>
      </el-form-item>
    </el-form>
  </div>
</template>

<script lang="ts" setup>
  import { updateProjSetUplist, type RequestAddProjSetUplistItem } from '@/api/masterBudget/projectRepo'

  import { useValidateArgs } from '@/hooks/useArgs'
  import { useReactive, useValue } from '@/hooks/useObject'
  import { useRules } from '@/hooks/useRules'
  import { useTableForm } from '@/hooks/useTable'
  import { useState } from '@/hooks/useState'
  import { useSuccess } from '@/hooks/useTip'

  const route = useRoute()
  const ELForm = ref()

  const state = useState({
    loading: false,
    id: ''
  })

  /** 表单数据 */
  const [form, _] = useReactive({
    items: <any>[]
  })

  /** 基础校验规则（模板级） */
  const rules = useRules([], {
    items: [
      {
        type: 'array',
        required: true,
        message: '请至少添加一条数据',
        trigger: ['change']
      }
    ]
  })

  /** 单元格校验规则 */
  const cellRules = useRules([{ key: 'name', label: '名称' }])

  /** 表格表单工具：getCellRules + renderRequiredHeader */
  const { getCellRules, renderRequiredHeader } = useTableForm(cellRules)

  /** 新增一行 */
  const cellTemp: () => RequestAddProjSetUplistItem = () => ({
    id: state.id,
    code: '',
    name: '',
    assetTypeId: ''
  })

  const handleAdd = () => {
    form.items.push(cellTemp())
  }

  /** 删除一行 */
  const handleDelete = (index: number) => {
    form.items.splice(index, 1)
  }

  /** 保存数据并校验 */
  const handleSave = () => {
    state.request({
      function: async () => {
        const request = useValue(form.items)
        await updateProjSetUplist(request)
        useSuccess('提交成功')
        return Promise.resolve()
      },
      showLoading: true,
      loadingTitle: `正在提交...`,
      ELForm,
    })
  }

  onMounted(async () => {
    const id = await useValidateArgs('id', true)
    state.id = id
  })
</script>

<style scoped>
  /* 表格内表单项样式调整 */
  .table-form-item {
    margin-bottom: 0;
  }

  .table-form-item :deep(.el-form-item__error) {
    position: absolute;
    bottom: -18px;
    left: 0;
  }
</style>

```
