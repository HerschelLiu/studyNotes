<template>
  <comp-drawer
    v-model="show"
    :z-index="zIndex"
    :enable-scroll="false"
    title="预算类别"
    :tip="tip"
    @save="handleSave">
    <div class="app-container">
      <!-- 搜索条件 start -->
      <CompTableSearch
        :model="list.query"
        :loading="list.loading"
        @search="handleSearch"
        @reset="handleReset">
        <template #left>
          <el-form-item label="类别名称">
            <el-input
              v-model.trim="list.query.budgetTypeName"
              placeholder="类别名称"
              clearable
              size="default"
              class="width-240" />
          </el-form-item>
        </template>

        <template #right>
          <el-text class="mr10">{{ checkAll.selected.value.length }} / {{ maxLength }}</el-text>
          <template v-if="maxLength > 1">
            <el-button-group>
              <el-button
                size="default"
                type="danger"
                @click="handleRemove">
                清除所有
              </el-button>
              <el-button
                v-if="isChecked"
                size="default"
                type="danger"
                @click="handleRemovePage">
                清除本页
              </el-button>
              <el-button
                v-else
                size="default"
                @click="handleSelectedPage">
                全选本页
              </el-button>
            </el-button-group>
          </template>
        </template>
      </CompTableSearch>
      <!-- 搜索条件 end -->

      <!-- 表格 start -->
      <el-table
        v-loading="list.loading"
        :data="list.items"
        element-loading-text="正在加载..."
        style="width: 100%"
        border>
        <el-table-column
          type="index"
          label="序号"
          width="80"
          align="center" />

        <el-table-column label="类别编码">
          <template #default="{ row }">
            {{ useFormatTable(row.code) }}
          </template>
        </el-table-column>

        <el-table-column label="类别名称">
          <template #default="{ row }">{{ useFormatTable(row.name) }}</template>
        </el-table-column>

        <el-table-column
          label="操作"
          class-name="small-padding"
          width="100">
          <template #default="{ row }">
            <el-button
              v-if="!includesRow(row)"
              type="primary"
              size="default"
              link
              :disabled="disabledRow(row)"
              @click="handleChecked(row, 1)">
              选择
            </el-button>
            <el-button
              v-else
              size="default"
              type="warning"
              link
              @click="handleChecked(row, 2)">
              取消
            </el-button>
          </template>
        </el-table-column>
      </el-table>
      <!-- 表格 end -->

      <pagination
        v-show="list.total > 0"
        :total="list.total"
        v-model:page="list.query.pageNum"
        v-model:limit="list.query.pageSize"
        @pagination="getList" />
    </div>
  </comp-drawer>
</template>

<script lang="ts" setup>
  import type {
    RequestListBudgetType,
    ResponseListBudgetType,
  } from '@/api/masterBudget/budgetInfo'

  import { listBudgetType } from '@/api/masterBudget/budgetInfo'
  import { putSelectedList, useCheckAll } from '@/hooks/useCheckAll'
  import { useList } from '@/hooks/useList'
  import { useClone, useValue } from '@/hooks/useObject'
  import { useSelect } from '@/hooks/useSelect'
  import { useFormatTable } from '@/hooks/useFormatTable'

  import CompDrawer from '@/components/Drawer/index.vue'
  import CompTableSearch from '@/components/TableSearch/index.vue'

  export type RequestList = RequestListBudgetType
  export type ResponseList = ResponseListBudgetType

  interface Props {
    /** 描述标题 */
    tip?: string
    /** 指定弹窗层级 */
    zIndex?: number
    selected: Partial<ResponseList>[]
    /** 最小选择数据 */
    minLength?: number
    /** 最大选择数据 */
    maxLength?: number
    /** 禁用选择行 */
    disabledRow?: (row: ResponseList) => boolean
  }

  const props = withDefaults(defineProps<Props>(), {
    zIndex: 1,
    selected: [],
    minLength: 1,
    maxLength: 500,
    disabledRow: () => false,
  })
  const emit = defineEmits(['change'])
  const show = defineModel<boolean>()

  // 表格多选依据字段
  const keys = 'id'

  const searchData = {
    budgetTypeName: '',
    status: 1,
  }
  const list = useList<RequestList, ResponseList>(useClone(searchData))

  /** 跨页选择 */
  const checkAll = useCheckAll<RequestList, Partial<ResponseList>>(keys)
  /** 选择弹窗hook */
  const {
    isChecked,
    handleChecked,
    handleRemove,
    handleRemovePage,
    handleSave,
    includesRow,
  } = useSelect<RequestList, Partial<ResponseList>>(props, { emit }, checkAll, list, keys)

  /** 初始化 */
  watch(
    () => props.modelValue,
    async val => {
      if (val) {
        checkAll.selected.value = props.selected
        if (!list.items.length) handleSearch()
      }
    }
  )

  const getList = () => {
    list.request(async () => {
      const data = await listBudgetType(useValue(list.query))
      const resData = checkAll.isCheckedItem(data.rows || [])

      return resData
    })
  }

  /** 全选本页 */
  const handleSelectedPage = () => {
    checkAll.selected.value = putSelectedList(
      checkAll.selected.value,
      list.items.filter(item => !props.disabledRow(item)),
      keys
    )
  }

  /** 搜索 */
  const handleSearch = () => {
    list.query.pageNum = 1
    getList()
  }

  /** 重置 */
  const handleReset = () => {
    list.query = useClone(searchData)
    handleSearch()
  }
</script>

    }

<template>
	<!-- 选择预算类别 -->
  <CompSelectBudgetType
    v-model="modalList.showSelectBudgetType"
    :selected="selectedBudgetType"
    @change="handleSelectedBudgetType" />
</template>

<script lang="ts" setup>
  const form = reactive({
    sampleTypeDtos: []
  })
	// 已选择的预算类别
  const selectedBudgetType = computed(() => {
    // map中组装的数据（如id）是选择弹窗中表格数据的key
    return form.sampleTypeDtos.map(item => ({ id: item.id })
  })
  
  // 选择预算类别
  const handleSelectedBudgetType = (rows: ResponseListBudgetType[]) => {
    form.sampleTypeDtos = rows
  }
</script>
