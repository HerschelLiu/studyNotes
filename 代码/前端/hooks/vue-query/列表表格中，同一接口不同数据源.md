在列表的form表单中，有两个下拉列表数据用的同一接口，第二个数据是根据前一个id查询出的，此种情况用vuequery如何写?

```vue
<template>
  <el-table
    :data="list"
    style="width: 100%"
    border
    show-summary
    :summary-method="getSummaries">
    

    <el-table-column
      label="转入科室"
      prop="inDeptId"
      width="200"
      :renderHeader="renderRequiredHeader('转入科室', 'inDeptId')">
      <template #default="{ row, $index }">
        <el-form-item
          label-width="0"
          :prop="`transferDetails.${$index}.inDeptId`"
          :rules="getCellRules('inDeptId')"
          inline-message>
          <el-select
            v-model="row.inDeptId"
            placeholder="请选择科室"
            filterable>
            <template #label="{ label }">{{ label }}</template>
            <el-option
              v-for="opt in deptList"
              :key="opt.inDeptId"
              :label="`${opt.inDeptCode} ${opt.inDeptName}`"
              :value="opt.inDeptId" />
          </el-select>
        </el-form-item>
      </template>
    </el-table-column>

    <el-table-column
      label="转入收支项目"
      prop="inIncomeExpenseId"
      min-width="200"
      :renderHeader="renderRequiredHeader('转入收支项目', 'inIncomeExpenseId')">
      <template #default="{ row, $index }">
        <el-form-item
          label-width="0"
          :prop="`transferDetails.${$index}.inIncomeExpenseId`"
          :rules="getCellRules('inIncomeExpenseId')"
          inline-message>
          <el-select
            v-model="row.inIncomeExpenseId"
            placeholder="请选择收支项目"
            filterable
            :disabled="!row.inDeptId">
            <template #label="{ label }">{{ label }}</template>
            <el-option
              v-for="opt in getRowProjectList($index)"
              :key="opt.inIncomeExpenseId"
              :label="`${opt.inIncomeExpenseCode} ${opt.inIncomeExpenseName}`"
              :value="opt.inIncomeExpenseId" />
          </el-select>
        </el-form-item>
      </template>
    </el-table-column>
  </el-table>
</template>

<script lang="ts" setup>
  import { listBudgetFillSummary } from '@/api/masterBudget/budgetPrep'
  import { keepPreviousData } from '@tanstack/vue-query'

  const list = defineModel<BudgBudgetTransferDetailVo[]>('list', {
    default: () => [],
  })

  /** 转入科室/收支项目公共接口 */
  const fetchSummaryRows = (query: RequestListBudgetFillSummary) =>
    listBudgetFillSummary(query).then(({ rows }) => rows)

  /** 年度+预算类型公共查询参数 */
  const baseSummaryQuery = computed<RequestListBudgetFillSummary>(() => ({
    yearId: props.form.yearId,
    budgetType: props.form.budgetType,
  }))

  /** 转入科室下拉 */
  const { data: deptListData } = useQuery({
    queryKey: computed(() => queryKeys.deptFill.listSummary(baseSummaryQuery.value)),
    queryFn: () => fetchSummaryRows(baseSummaryQuery.value),
    enabled: computed(() => !!props.form.yearId && !!props.form.budgetType),
  })
  const deptList = computed(() => deptListData.value ?? [])

  /** 按行动态查询转入收支项目列表（每行基于该行 inDeptId 独立查询，互不影响） */
  const rowProjectResults = useQueries({
    queries: computed(() =>
      list.value.map(row => {
        const query: RequestListBudgetFillSummary = {
          ...baseSummaryQuery.value,
          compileDeptId: row.inDeptId,
        }
        return {
          queryKey: queryKeys.deptFill.listSummary(query),
          queryFn: () => fetchSummaryRows(query),
          enabled: !!row.inDeptId,
        }
      })
    ),
  })
  /** 获取第 index 行的转入收支项目列表 */
  const getRowProjectList = (index: number) => rowProjectResults.value[index]?.data
</script>

```
